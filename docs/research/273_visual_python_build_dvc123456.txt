class VisualPythonVM:
    def __init__(self):
        self.stack = []
        self.namespace = {}
        self.execution_log = []
        self.current_ip = 0  # Instruction pointer
        self.last_result = None  # For Ans
        
    def execute_bytecode(self, instructions):
        """Execute Python bytecode instructions with support for jumps and 3.11+ opcodes"""
        output = io.StringIO()
        old_stdout = sys.stdout
        sys.stdout = output
        
        # Build offset to index mapping for jumps
        offset_to_index = {ins.offset: i for i, ins in enumerate(instructions)}
        
        while self.current_ip < len(instructions):
            instr = instructions[self.current_ip]
            
            # Log the instruction for visualization
            self.execution_log.append({
                'instruction': instr.opname,
                'arg': instr.argval if hasattr(instr, 'argval') else instr.arg,
                'stack_before': list(self.stack),
                'ip': self.current_ip
            })
            
            # Execute the instruction
            self.execute_instruction(instr, instructions, offset_to_index)
            
        sys.stdout = old_stdout
        return output.getvalue()
    
    def jump_to_offset(self, target_offset, instructions, offset_to_index):
        """Helper function to jump to a specific offset"""
        return offset_to_index.get(target_offset, len(instructions))
    
    def execute_instruction(self, instruction, instructions, offset_to_index):
        """Execute a single instruction with support for 3.11+ opcodes and jumps"""
        opname = instruction.opname
        arg = instruction.argval if hasattr(instruction, 'argval') else instruction.arg
        
        # Handle comparison operations
        if opname == 'COMPARE_OP':
            b, a = self.stack.pop(), self.stack.pop()
            cmp = arg  # In 3.11+, argval is a string like '==', '<', etc.
            self.stack.append({
                '==': a == b, '!=': a != b,
                '<': a < b,  '<=': a <= b,
                '>': a > b,  '>=': a >= b,
                'in': a in b, 'not in': a not in b,
                'is': a is b, 'is not': a is not b
            }[cmp])
            self.current_ip += 1
            
        # Handle conditional jumps
        elif opname in ("POP_JUMP_FORWARD_IF_FALSE", "POP_JUMP_BACKWARD_IF_FALSE"):
            cond = self.stack.pop()
            if not cond:
                self.current_ip = self.jump_to_offset(arg, instructions, offset_to_index)
            else:
                self.current_ip += 1
                
        elif opname in ("POP_JUMP_FORWARD_IF_TRUE", "POP_JUMP_BACKWARD_IF_TRUE"):
            cond = self.stack.pop()
            if cond:
                self.current_ip = self.jump_to_offset(arg, instructions, offset_to_index)
            else:
                self.current_ip += 1
                
        # Handle unconditional jumps
        elif opname in ("JUMP_FORWARD", "JUMP_BACKWARD", "JUMP_ABSOLUTE"):
            self.current_ip = self.jump_to_offset(arg, instructions, offset_to_index)
            
        # Handle 3.11+ compact arithmetic
        elif opname == "BINARY_OP":
            b, a = self.stack.pop(), self.stack.pop()
            opstr = arg  # e.g. '+', '-', '*', '/', '//', '%', '**'
            self.stack.append({
                '+': a + b, '-': a - b, '*': a * b, '/': a / b,
                '//': a // b, '%': a % b, '**': a ** b,
                '@': a @ b if hasattr(a, '__matmul__') else (_ for _ in ()).throw(TypeError("@ not supported"))
            }[opstr])
            self.current_ip += 1
            
        # Handle 3.11+ function calls
        elif opname == "PUSH_NULL":
            self.stack.append(None)
            self.current_ip += 1
            
        elif opname == "PRECALL":
            # No-op for us, just advance
            self.current_ip += 1
            
        elif opname == "CALL":
            argc = arg  # number of positional args
            args = [self.stack.pop() for _ in range(argc)][::-1]
            func = self.stack.pop()
            if func is print:
                print(*args)
                self.last_result = args[0] if len(args) == 1 else args
                self.stack.append(None)
            else:
                result = func(*args)
                self.last_result = result
                self.stack.append(result)
            self.current_ip += 1
            
        # Original handlers for compatibility
        elif opname == 'LOAD_CONST':
            self.stack.append(arg)
            self.current_ip += 1
            
        elif opname == 'LOAD_NAME':
            if arg == 'Ans' and self.last_result is not None:
                self.stack.append(self.last_result)
            else:
                self.stack.append(self.namespace.get(arg, None))
            self.current_ip += 1
            
        elif opname == 'STORE_NAME':
            self.namespace[arg] = self.stack.pop()
            self.current_ip += 1
            
        elif opname == 'BINARY_ADD':
            b, a = self.stack.pop(), self.stack.pop()
            self.stack.append(a + b)
            self.current_ip += 1
            
        elif opname == 'BINARY_SUBTRACT':
            b, a = self.stack.pop(), self.stack.pop()
            self.stack.append(a - b)
            self.current_ip += 1
            
        elif opname == 'BINARY_MULTIPLY':
            b, a = self.stack.pop(), self.stack.pop()
            self.stack.append(a * b)
            self.current_ip += 1
            
        elif opname == 'BINARY_TRUE_DIVIDE':
            b, a = self.stack.pop(), self.stack.pop()
            self.stack.append(a / b)
            self.current_ip += 1
            
        elif opname == 'CALL_FUNCTION':
            func = self.stack.pop()
            args = [self.stack.pop() for _ in range(arg)]
            result = func(*reversed(args))
            self.last_result = result
            self.stack.append(result)
            self.current_ip += 1
            
        elif opname == 'POP_TOP':
            self.stack.pop()
            self.current_ip += 1

def code_to_visual_tiles(code_obj):
    """Convert Python bytecode to visual tiles"""
    instructions = list(dis.get_instructions(code_obj))
    tiles = []
    
    for instr in instructions:
        if instr.opname in VISUAL_OPCODE_MAP:
            tile_info = VISUAL_OPCODE_MAP[instr.opname].copy()
            tile_info['opname'] = instr.opname
            if hasattr(instr, 'argval') and instr.argval is not None:
                tile_info['arg'] = str(instr.argval)
            else:
                tile_info['arg'] = ''
            tile_info['offset'] = instr.offset
            tile_info['is_jump_target'] = instr.is_jump_target
            tiles.append(tile_info)
    
    return tiles

            expr = expr.strip()

            result = self._evaluate_expression(expr)

            '___ans___': self.vm.last_result,
            **self.vm.namespace,
            **self.allowed_names
        }

            # Compile and execute the expression

class CalculatorVM:
    def __init__(self):
        self.stack: List[Any] = []
        self.variables: Dict[str, Any] = {}
        self.execution_log: List[Dict] = []
        self.history: List[Dict[str, str]] = []

    def execute_bytecode(self, instructions: List[dis.Instruction]) -> str:
        """Execute bytecode in a sandboxed environment."""
        output = io.StringIO()
        old_stdout = sys.stdout
        sys.stdout = output
        offset_to_index = {ins.offset: i for i, ins in enumerate(instructions)}
        ip = 0
        try:
            while ip < len(instructions):
                instr = instructions[ip]
                self.execution_log.append({"instr": instr.opname, "arg": str(instr.argval), "stack": list(self.stack), "ip": ip})
                if instr.opname == "LOAD_CONST":
                    self.stack.append(instr.argval)
                    ip += 1
                elif instr.opname == "LOAD_NAME":
                    name = instr.argval
                    if name == "Ans":
                        self.stack.append(self.ans)
                    elif name in ALLOWED:
                        self.stack.append(ALLOWED[name])
                    else:
                        self.stack.append(self.variables.get(name, None))
                    ip += 1
                elif instr.opname == "STORE_NAME":
                    self.variables[instr.argval] = self.stack.pop()
                    ip += 1
                elif instr.opname == "BINARY_OP":
                    b, a = self.stack.pop(), self.stack.pop()
                    opstr = instr.argval
                    self.stack.append({
                        '+': a + b, '-': a - b, '*': a * b, '/': a / b,
                        '//': a // b, '%': a % b, '**': a ** b
                    }[opstr])
                    ip += 1
                elif instr.opname == "BINARY_ADD":
                    b, a = self.stack.pop(), self.stack.pop()
                    self.stack.append(a + b)
                    ip += 1
                elif instr.opname == "BINARY_MULTIPLY":
                    b, a = self.stack.pop(), self.stack.pop()
                    self.stack.append(a * b)
                    ip += 1
                elif instr.opname == "PRECALL":
                    ip += 1
                elif instr.opname == "CALL":
                    argc = instr.arg
                    args = [self.stack.pop() for _ in range(argc)][::-1]
                    func = self.stack.pop()
                    if func == print:
                        print(*args)
                        self.stack.append(None)
                    else:
                        self.stack.append(func(*args))
                    ip += 1
                elif instr.opname == "POP_TOP":
                    self.stack.pop()
                    ip += 1
                elif instr.opname == "RETURN_VALUE":
                    if self.stack:
                        self.ans = self.stack.pop()
                    break
                else:
                    raise NotImplementedError(f"Unsupported opcode: {instr.opname}")
        except Exception as e:
            self.last_output = f"Error: {e}"
            return self.last_output
        finally:
            self.last_output = output.getvalue().strip()
            sys.stdout = old_stdout
        return self.last_output

    """Convert Python source to bytecode instructions."""

def decode_tiles(instructions: List[dis.Instruction]) -> List[tuple]:
    """Decode tiles for drift detection."""
    return [(instr.opname, str(instr.argval) if instr.argval else "") for instr in instructions]

def compute_crc(data: str) -> str:
    """Compute CRC for source code."""
    return hashlib.md5(data.encode()).hexdigest()[:8]

VISUAL_OPCODE_MAP = {
    'LOAD_CONST': {'color': '#3B82F6', 'symbol': '⚡'},
    'LOAD_NAME': {'color': '#10B981', 'symbol': '📛'},
    'STORE_NAME': {'color': '#F59E0B', 'symbol': '💾'},
    'BINARY_ADD': {'color': '#EF4444', 'symbol': '➕'},
    'BINARY_SUBTRACT': {'color': '#8B5CF6', 'symbol': '➖'},

    'CALL_FUNCTION': {'color': '#F97316', 'symbol': '📞'},
    'PRECALL': {'color': '#A855F7', 'symbol': '⏳'},

    'COMPARE_OP': {'color': '#8B5CF6', 'symbol': '⚖️'},
    'POP_JUMP_FORWARD_IF_FALSE': {'color': '#F43F5E', 'symbol': '↩️'},
    'POP_JUMP_BACKWARD_IF_FALSE': {'color': '#F43F5E', 'symbol': '↩️'},
    'POP_JUMP_FORWARD_IF_TRUE': {'color': '#F43F5E', 'symbol': '↪️'},
    'POP_JUMP_BACKWARD_IF_TRUE': {'color': '#F43F5E', 'symbol': '↪️'},

    'GET_ITER': {'color': '#A855F7', 'symbol': '🔄'},

class VisualPythonVM:
    def __init__(self):
        self.stack = []
        self.namespace = {'Ans': 0}  # Initialize with 'Ans' variable
        self.execution_log = []
        self.current_ip = 0
        self.allowed_globals = {
            'abs': abs, 'round': round, 'min': min, 'max': max, 'pow': pow,
            'sum': sum, 'len': len, 'range': range, 'print': print
        }
        # Add math module functions
        import math
        self.allowed_globals.update({k: getattr(math, k) for k in dir(math) if not k.startswith('_')})

    def execute_bytecode(self, instructions):
        """Execute Python bytecode instructions with support for jumps and 3.11+ opcodes"""
        output = io.StringIO()
        old_stdout = sys.stdout
        sys.stdout = output

        # Build offset to index mapping for jumps
        offset_to_index = {ins.offset: i for i, ins in enumerate(instructions)}
        self.current_ip = 0

        while self.current_ip < len(instructions):
            instr = instructions[self.current_ip]

            # Log the instruction for visualization
            self.execution_log.append({
                'instruction': instr.opname,
                'arg': instr.argval if hasattr(instr, 'argval') else instr.arg,
                'stack_before': list(self.stack),
                'ip': self.current_ip
            })

            # Execute the instruction
            self.execute_instruction(instr, instructions, offset_to_index)

        sys.stdout = old_stdout
        return output.getvalue()

    def jump_to_offset(self, target_offset, instructions, offset_to_index):
        """Helper function to jump to a specific offset"""
        return offset_to_index.get(target_offset, len(instructions))

    def execute_instruction(self, instruction, instructions, offset_to_index):
        """Execute a single instruction with support for 3.11+ opcodes and jumps"""
        opname = instruction.opname
        arg = instruction.argval if hasattr(instruction, 'argval') else instruction.arg

        # Handle comparison operations
        if opname == 'COMPARE_OP':
            b, a = self.stack.pop(), self.stack.pop()
            cmp = arg  # In 3.11+, argval is a string like '==', '<', etc.
            self.stack.append({
                '==': a == b, '!=': a != b,
                '<': a < b, '<=': a <= b,
                '>': a > b, '>=': a >= b,
                'in': a in b, 'not in': a not in b,
                'is': a is b, 'is not': a is not b
            }[cmp])
            self.current_ip += 1

        # Handle conditional jumps
        elif opname in ("POP_JUMP_FORWARD_IF_FALSE", "POP_JUMP_BACKWARD_IF_FALSE"):
            cond = self.stack.pop()
            if not cond:
                self.current_ip = self.jump_to_offset(arg, instructions, offset_to_index)
            else:
                self.current_ip += 1

        elif opname in ("POP_JUMP_FORWARD_IF_TRUE", "POP_JUMP_BACKWARD_IF_TRUE"):
            cond = self.stack.pop()
            if cond:
                self.current_ip = self.jump_to_offset(arg, instructions, offset_to_index)
            else:
                self.current_ip += 1

        # Handle unconditional jumps
        elif opname in ("JUMP_FORWARD", "JUMP_BACKWARD", "JUMP_ABSOLUTE"):
            self.current_ip = self.jump_to_offset(arg, instructions, offset_to_index)

        # Handle 3.11+ compact arithmetic
        elif opname == "BINARY_OP":
            b, a = self.stack.pop(), self.stack.pop()
            opstr = arg  # e.g. '+', '-', '*', '/', '//', '%', '**'
            self.stack.append({
                '+': a + b, '-': a - b, '*': a * b, '/': a / b,
                '//': a // b, '%': a % b, '**': a ** b,
                '@': a @ b if hasattr(a, '__matmul__') else (_ for _ in ()).throw(TypeError("@ not supported"))
            }[opstr])
            self.current_ip += 1

        # Handle 3.11+ function calls
        elif opname == "PRECALL":
            self.current_ip += 1

        elif opname == "CALL":
            argc = arg  # number of positional args
            args = [self.stack.pop() for _ in range(argc)][::-1]
            func = self.stack.pop()
            if func is print:
                print(*args)
                self.stack.append(None)
            elif func in self.allowed_globals:
                self.stack.append(func(*args))
            else:
                self.stack.append(None)
            self.current_ip += 1

        # Original handlers for compatibility
        elif opname == 'LOAD_CONST':
            self.stack.append(arg)
            self.current_ip += 1

        elif opname == 'LOAD_NAME':
            if arg in self.namespace:
                self.stack.append(self.namespace[arg])
            elif arg in self.allowed_globals:
                self.stack.append(self.allowed_globals[arg])
            else:
                self.stack.append(None)
            self.current_ip += 1

        elif opname == 'STORE_NAME':
            self.namespace[arg] = self.stack.pop()
            self.current_ip += 1

        elif opname == 'BINARY_ADD':
            b, a = self.stack.pop(), self.stack.pop()
            self.stack.append(a + b)
            self.current_ip += 1

        elif opname == 'BINARY_SUBTRACT':
            b, a = self.stack.pop(), self.stack.pop()
            self.stack.append(a - b)
            self.current_ip += 1

        elif opname == 'BINARY_MULTIPLY':
            b, a = self.stack.pop(), self.stack.pop()
            self.stack.append(a * b)
            self.current_ip += 1

        elif opname == 'BINARY_TRUE_DIVIDE':
            b, a = self.stack.pop(), self.stack.pop()
            self.stack.append(a / b)
            self.current_ip += 1

        elif opname == 'POP_TOP':
            self.stack.pop()
            self.current_ip += 1

        else:
            # graceful fallback for unhandled opcodes
            self.current_ip += 1

def to_visual_ir(py_src: str):
    """Convert Python source code to visual IR (bytecode instructions)"""
    try:
        code = compile(py_src, "<visual>", "exec")
        return list(dis.get_instructions(code))
    except Exception as e:
        return [{"opname": "ERROR", "argval": str(e)}]

def decode_tiles(instructions):
    """Decode tiles back to bytecode for drift detection"""
    return [(instr.opname, str(instr.argval) if hasattr(instr, 'argval') else "") for instr in instructions]

def compute_crc(data: str):
    """Compute a CRC for verification"""
    return hashlib.md5(data.encode()).hexdigest()[:8]

def get_tile_color(item: str, is_opcode: bool = True) -> str:
    """Get color for a tile based on opcode or value"""

    return VISUAL_OPCODE_MAP.get("CONST" if isinstance(item, (int, str)) else "NAME", {}).get('color', "#64748b")

To achieve this, we'll follow a strategic blueprint that treats your visual VM as the unassailable engine and wraps it in a shell that can handle a user's input. The architecture can be broken down into three core components: the "CalcVM," the "Execution Sandbox," and the "Visualizer."

### 1. The CalcVM: A Specialized Interpreter

At its heart, the calculator is nothing more than a simplified version of the Python Virtual Machine you've already built. It's a stack-based machine that operates on a "low-level, platform-independent set of instructions".[1] To transform it into a calculator engine, we will make two key adjustments.

First, we will expand its instruction set to handle the precise needs of mathematical operations. Your `a*b` demo already has `BINARY_MULTIPLY`, but a true calculator needs more. It requires support for all standard arithmetic and comparison operations. Modern Python bytecode consolidates these into a single `BINARY_OP` opcode with a distinguishing argument, and we can implement a robust handler for this opcode to perform everything from addition (`+`) to modulo (`%`).[2]

Second, we'll implement the necessary control flow. This is where the magic of conditionals and jumps comes in. The VM will need to handle `COMPARE_OP` for logical comparisons (`==`, `<=`, `>`), and then use jump opcodes like `POP_JUMP_FORWARD_IF_FALSE` to navigate conditional expressions.[3] This allows for complex calculator logic such as `if` statements or even a `while` loop to compute a sum.

### 2. The Execution Sandbox: A Safe Environment

A real-world application requires a secure environment. We will contain the VM within a tightly controlled sandbox, a crucial step to prevent malicious code from being executed. This involves running the VM in a restricted namespace that only includes a "whitelisted" set of functions, such as `math` library operations (`sqrt`, `sin`, `cos`) and other safe built-ins (`abs`, `min`, `max`).[4] This ensures that user input cannot access arbitrary system functions or variables outside of the designated environment. Any variables defined by the user (e.g., `x = 42`) will be stored within the sandbox's namespace, and a special `Ans` variable will be implemented to hold the result of the previous calculation.

### 3. The Visualizer: The Explanatory Layer

The third piece is the interactive front end. The most direct approach is a Command Line Interface (CLI) that acts as a Read-Eval-Print Loop (REPL). It will:
- **Read:** Accept a line of user input, like `2+3*4`.
- **Eval:** Compile the input into bytecode and pass it to the CalcVM for execution.
- **Print:** Display the final result to the user.
- **Loop:** Repeat the process, maintaining the variable state.

The beauty of this design is that the existing 4-pane artifact can serve as the ultimate debugging and educational tool. After each calculation, the REPL can be instructed to generate a new 4-pane artifact, effectively creating a "visual tape calculator." This allows you to step back and examine exactly how the VM processed the expression. You could see the bytecode tiles for `2`, `3`, `*`, and `4` and then watch in Pane 4 as the VM's stack performs the operations and produces the final result of `14`. This makes the inner workings of the calculator transparent and verifiable.

import dis
import io
import sys
from math import *
from types import SimpleNamespace
from PIL import Image, ImageDraw, ImageFont

# ------------------------------
# Safe environment (whitelist)
# ------------------------------
def _whitelist_math():
    import math
    allowed = {k: getattr(math, k) for k in dir(math) if not k.startswith("_")}
    # Handy builtins
    allowed.update({
        "abs": abs, "round": round, "min": min, "max": max, "pow": pow,
        "int": int, "float": float, "complex": complex,
        # common constants already included from math, but ensure presence:
        "pi": math.pi, "e": math.e, "tau": math.tau,
    })
    return allowed

ALLOWED = _whitelist_math()

class VisualCalcVM:
    """Tiny bytecode VM tuned for calculator use (Python 3.11+ friendly)."""
    def __init__(self):
        self.stack = []
        self.vars = {"Ans": 0}
        self.exec_log = []  # for visualization

    def _jump_to(self, target_offset, offset_to_index, nins):
        return offset_to_index.get(target_offset, nins)

    def run(self, code_str, mode="auto"):
        """Execute code_str. Returns (value, stdout_str).
        mode: 'eval', 'exec', or 'auto' (detect).
        """
        # Detect mode
        detected_mode = mode
        if mode == "auto":
            detected_mode = "eval" if self._looks_like_expr(code_str) else "exec"

        # Compile
        co = compile(code_str, "<calc>", detected_mode)
        instructions = list(dis.get_instructions(co))
        offset_to_index = {ins.offset: i for i, ins in enumerate(instructions)}
        ip = 0
        nins = len(instructions)

        # Prepare stdout capture
        old_stdout = sys.stdout
        sys.stdout = io.StringIO()

        try:
            while ip < nins:
                ins = instructions[ip]
                op = ins.opname
                self.exec_log.append((ip, op, ins.argval))

                # --- No-ops / housekeeping for 3.11 ---
                if op in ("RESUME", "CACHE", "EXTENDED_ARG"):
                    ip += 1

                # --- Constants & names ---
                elif op == "LOAD_CONST":
                    self.stack.append(ins.argval); ip += 1
                elif op in ("LOAD_NAME", "LOAD_GLOBAL"):
                    name = ins.argval
                    if name in self.vars:
                        self.stack.append(self.vars[name])
                    elif name in ALLOWED:
                        self.stack.append(ALLOWED[name])
                    elif name == "__builtins__":  # some code objects may try to load this; block
                        raise NameError("access to __builtins__ is not allowed")
                    else:
                        raise NameError(f"name '{name}' is not defined (or not allowed)")
                    ip += 1
                elif op == "STORE_NAME":
                    self.vars[ins.argval] = self.stack.pop(); ip += 1

                # --- Unary ops ---
                elif op == "UNARY_NEGATIVE":
                    self.stack.append(-self.stack.pop()); ip += 1
                elif op == "UNARY_POSITIVE":
                    self.stack.append(+self.stack.pop()); ip += 1
                elif op == "UNARY_NOT":
                    self.stack.append(not self.stack.pop()); ip += 1

                # --- Binary ops (3.11 unified) ---
                elif op == "BINARY_OP":
                    b, a = self.stack.pop(), self.stack.pop()
                    sym = ins.argrepr  # operator symbol, e.g. "+", "-", "*", "/", "//", "%", "**", "@"
                    if sym == '+':   self.stack.append(a + b)
                    elif sym == '-': self.stack.append(a - b)
                    elif sym == '*': self.stack.append(a * b)
                    elif sym == '/': self.stack.append(a / b)
                    elif sym == '//': self.stack.append(a // b)
                    elif sym == '%': self.stack.append(a % b)
                    elif sym == '**': self.stack.append(a ** b)
                    elif sym == '@':
                        if hasattr(a, '__matmul__'): self.stack.append(a @ b)
                        else: raise TypeError("@ not supported for operands")
                    else:
                        raise NotImplementedError(f"BINARY_OP {sym}")
                    ip += 1
                # Legacy add/mul for older versions (harmless if unused)
                elif op == "BINARY_ADD":
                    b, a = self.stack.pop(), self.stack.pop(); self.stack.append(a + b); ip += 1
                elif op == "BINARY_MULTIPLY":
                    b, a = self.stack.pop(), self.stack.pop(); self.stack.append(a * b); ip += 1

                # --- Comparisons ---
                elif op == "COMPARE_OP":
                    b, a = self.stack.pop(), self.stack.pop()
                    cmp = ins.argrepr
                    res = {
                        '==': a == b, '!=': a != b, '<': a < b,  '<=': a <= b,
                        '>': a > b,  '>=': a >= b, 'in': a in b, 'not in': a not in b,
                        'is': a is b, 'is not': a is not b
                    }[cmp]
                    self.stack.append(res); ip += 1

                # --- Jumps ---
                elif op in ("POP_JUMP_FORWARD_IF_FALSE", "POP_JUMP_BACKWARD_IF_FALSE"):
                    cond = self.stack.pop()
                    if not cond:
                        ip = self._jump_to(ins.argval, offset_to_index, nins)
                    else:
                        ip += 1
                elif op in ("POP_JUMP_FORWARD_IF_TRUE", "POP_JUMP_BACKWARD_IF_TRUE"):
                    cond = self.stack.pop()
                    if cond:
                        ip = self._jump_to(ins.argval, offset_to_index, nins)
                    else:
                        ip += 1
                elif op in ("JUMP_FORWARD", "JUMP_BACKWARD", "JUMP_ABSOLUTE"):
                    ip = self._jump_to(ins.argval, offset_to_index, nins)

                # --- Calls (3.11 pattern) ---
                elif op == "PUSH_NULL":
                    self.stack.append(None); ip += 1
                elif op == "PRECALL":
                    ip += 1  # no-op
                elif op == "CALL":
                    argc = ins.arg
                    args = [self.stack.pop() for _ in range(argc)][::-1]
                    func = self.stack.pop()
                    # Only allow whitelisted functions or user-defined callables in vars
                    if func in ALLOWED.values() or any((k for k,v in self.vars.items() if v is func)):
                        result = func(*args)
                        self.stack.append(result)
                    elif func is print:
                        print(*args); self.stack.append(None)
                    else:
                        raise PermissionError("function call not allowed in calculator")
                    ip += 1

                # --- Stack / return ---
                elif op == "POP_TOP":
                    if self.stack: self.stack.pop()
                    ip += 1
                elif op == "RETURN_VALUE":
                    break

                else:
                    raise NotImplementedError(f"Opcode not supported: {op}")
        finally:
            out = sys.stdout.getvalue()
            sys.stdout = old_stdout

        # Value to return: top of stack if present (eval mode), else None
        value = self.stack[-1] if self.stack else None
        self.vars["Ans"] = value if value is not None else self.vars.get("Ans", None)
        return value, out

    def _looks_like_expr(self, s):
        # Heuristic: if there's an '=' at top level, treat as exec; else eval
        # This is simplistic but fine for calculator usage
        return '=' not in s.strip().split('#', 1)[0]  # ignore comments

def render_4pane(expr, vm: VisualCalcVM, outfile):
    # Compile & disassemble
    co = compile(expr, "<calc-pane>", "eval" if vm._looks_like_expr(expr) else "exec")
    instructions = list(dis.get_instructions(co))

    # Execute via VM
    vm.exec_log.clear()
    val, out = vm.run(expr, mode="auto")

    W, H = 1400, 900
    img = Image.new("RGB", (W, H), "#0a0f1a")
    d = ImageDraw.Draw(img)
    try:
        font_code = ImageFont.truetype("DejaVuSansMono.ttf", 28)
        font_ui = ImageFont.truetype("DejaVuSans.ttf", 16)
        font_small = ImageFont.truetype("DejaVuSans.ttf", 14)
    except:
        font_code = ImageFont.load_default()
        font_ui = ImageFont.load_default()
        font_small = ImageFont.load_default()

    def pane(x, y, w, h, title):
        d.rounded_rectangle([x, y, x+w, y+h], radius=14, fill="#0b1220", outline="#1f2a44", width=2)
        d.text((x+14, y-24), title, fill="#a5b4fc", font=font_ui)
        return (x, y, x+w, y+h)

    P = []
    gap = 18
    w = (W - gap*3)//2
    h = (H - gap*3)//2
    P.append(pane(gap, gap*2, w, h, "Pane 1 — Source"))
    P.append(pane(gap*2+w, gap*2, w, h, "Pane 2 — Pixelized text"))
    P.append(pane(gap, gap*3+h, w, h, "Pane 3 — Bytecode tiles"))
    P.append(pane(gap*2+w, gap*3+h, w, h, "Pane 4 — Result"))

    # Pane 1: source
    x0, y0, x1, y1 = P[0]
    y = y0+20
    for line in expr.splitlines() or ['']:
        d.text((x0+20, y), line, fill="#e2e8f0", font=font_code)
        y += 34

    # Pane 2: pixelized text
    x0, y0, x1, y1 = P[1]
    cell_w, cell_h = 18, 24
    gx, gy = x0+16, y0+16
    # grid
    for r, line in enumerate(expr.splitlines() or ['']):
        for c, ch in enumerate(line):
            cx, cy = gx + c*cell_w, gy + r*cell_h
            d.rectangle([cx, cy, cx+cell_w-2, cy+cell_h-2], fill="#1e293b")
            d.text((cx+3, cy+2), ch, fill="#c7d2fe", font=font_small)

    # Pane 3: bytecode tiles
    x0, y0, x1, y1 = P[2]
    tile = 48
    cols = max(1, (w-40)//(tile+6))
    ox, oy = x0+20, y0+24
    for i, ins in enumerate(instructions):
        cx = ox + (i%cols)*(tile+6)
        cy = oy + (i//cols)*(tile+6)
        color = {
            "LOAD_CONST": "#3b82f6",
            "LOAD_NAME": "#a855f7",
            "LOAD_GLOBAL": "#a855f7",
            "STORE_NAME": "#f97316",
            "BINARY_OP": "#ef4444",
            "COMPARE_OP": "#8b5cf6",
            "PRECALL": "#94a3b8",
            "CALL": "#10b981",
            "POP_TOP": "#f59e0b",
            "RETURN_VALUE": "#94a3b8",
            "PUSH_NULL": "#64748b",
        }.get(ins.opname, "#475569")
        d.rounded_rectangle([cx, cy, cx+tile, cy+tile], radius=10, fill=color)
        lab = ins.opname if ins.argval is None else f"{ins.opname}\n{ins.argval}"
        d.text((cx+tile/2, cy+tile/2), lab, anchor="mm", fill="#ffffff", font=font_small)

    # Pane 4: results
    x0, y0, x1, y1 = P[3]
    y = y0+16

    d.text((x0+16, y), f"Ans = {vm.vars.get('Ans')}", fill="#22c55e", font=font_ui); y+=22

    img.save(outfile)

        try:
            val, out = vm.run(line, mode="auto")
            if out.strip(): print(out.strip())
            if val is not None:
                print(val)
            history.append((line, val))
