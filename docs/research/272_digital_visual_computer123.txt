py2px.py — minimal Python→pixels runtime (PIL backend, 60 Hz ticks, frame dumps)
#!/usr/bin/env python3
# py2px.py
from __future__ import annotations
from dataclasses import dataclass
from typing import Tuple, Optional, Iterable, Callable, List
from PIL import Image, ImageDraw, ImageFont
import os, time

# ---------- Core IR (immediate-mode ops) ----------
@dataclass
class Rect:  x:int; y:int; w:int; h:int; color:str
@dataclass
class Line:  x1:int; y1:int; x2:int; y2:int; width:int; color:str
@dataclass
class Text:  x:int; y:int; text:str; size:int; color:str
@dataclass
class Clear: color:str
@dataclass
class Sleep: frames:int                 # sleep N ticks
@dataclass
class Tick:  pass                        # end-of-frame marker (optional)

Op = Rect | Line | Text | Clear | Sleep | Tick

# ---------- Runtime ----------
class Px:
    def __init__(self, width:int=640, height:int=360, fps:int=60, out_dir:str|None="out"):
        self.width, self.height, self.fps = width, height, fps
        self.out_dir = out_dir
        self._frame_idx = 0
        self._img = Image.new("RGB", (width, height), "#000000")
        self._draw = ImageDraw.Draw(self._img)
        self._default_font = None  # load lazily
        if out_dir:
            os.makedirs(out_dir, exist_ok=True)

    # ---- Drawing ops (imperative, immediate-mode) ----
    def clear(self, color:str="#000000"):     self._apply(Clear(color))
    def rect(self, x:int,y:int,w:int,h:int,color:str): self._apply(Rect(x,y,w,h,color))
    def line(self,x1:int,y1:int,x2:int,y2:int,width:int=1,color:str="#FFFFFF"): self._apply(Line(x1,y1,x2,y2,width,color))
    def text(self,x:int,y:int,text:str,size:int=14,color:str="#FFFFFF"): self._apply(Text(x,y,text,size,color))
    def sleep(self, frames:int):              self._apply(Sleep(frames))
    def tick(self):                           self._apply(Tick())

    # ---- Batch support (if you want to yield ops) ----
    def run(self, ops: Iterable[Op]): 
        for op in ops: self._apply(op)

    # ---- Internals ----
    def _apply(self, op: Op):
        if isinstance(op, Clear):
            self._img.paste(Image.new("RGB", (self.width, self.height), op.color))
            self._draw = ImageDraw.Draw(self._img)
        elif isinstance(op, Rect):
            self._draw.rectangle([op.x, op.y, op.x+op.w-1, op.y+op.h-1], fill=op.color)
        elif isinstance(op, Line):
            self._draw.line([op.x1, op.y1, op.x2, op.y2], fill=op.color, width=op.width)
        elif isinstance(op, Text):
            font = self._font(op.size)
            self._draw.text((op.x, op.y), op.text, fill=op.color, font=font)
        elif isinstance(op, Sleep):
            for _ in range(max(0, op.frames)): self._commit()
        elif isinstance(op, Tick):
            self._commit()
        else:
            raise TypeError(f"Unknown op {op}")

    def _font(self, size:int):
        if self._default_font is None:
            try:
                self._default_font = ImageFont.load_default()
            except Exception:
                self._default_font = ImageFont.load_default()
        if hasattr(ImageFont, "truetype"):
            # Try a basic truetype if available; fallback otherwise
            try: return ImageFont.truetype("arial.ttf", size)
            except Exception: return self._default_font
        return self._default_font

    def _commit(self):
        self._frame_idx += 1
        if self.out_dir:
            self._img.save(os.path.join(self.out_dir, f"frame_{self._frame_idx:06d}.png"))

# ---------- Python-embedded DSL helpers ----------
def program(width:int=640, height:int=360, fps:int=60, out_dir:str|None="out"):
    """
    Decorator: turns a function f(px) into an entrypoint that renders frames.
    Usage:
        @program(320,240,60,"out")
        def main(px): ...
    """
    def wrap(fn: Callable[[Px], None]):
        def runner():
            px = Px(width, height, fps, out_dir)
            t0 = time.time()
            fn(px)
            # ensure at least one frame exists for static screens
            if px._frame_idx == 0: px.tick()
            dt = time.time()-t0
            print(f"[py2px] wrote {px._frame_idx} frames to {px.out_dir or '(memory)'} in {dt:.2f}s")
        return runner
    return wrap

# ---------- Demo program when run as a script ----------
@program(width=320, height=200, fps=60, out_dir="out_demo")
def main(px: Px):
    px.clear("#111111")
    px.text(8, 8, "PXOS — Python → Pixels", 12, "#A0FFA0")

# screen_base.py
from dataclasses import dataclass
from typing import Dict, Tuple, Protocol
from PIL import Image, ImageDraw, ImageStat

CELL = 4
CTRL_Y0 = 0
FIELDS = [("SCREEN_ID", 12), ("PC", 12), ("MAILBOX_IN", 8), ("MAILBOX_OUT", 8), ("CRC16", 16)]

def draw_control(draw: ImageDraw.ImageDraw, fields: Dict[str, int]):
    x = 4; y = CTRL_Y0 + 4
    for name, nbits in FIELDS:
        val = fields.get(name, 0)
        for b in range(nbits):
            bit = (val >> (nbits-1-b)) & 1
            x0 = x + b * CELL
            draw.rectangle([x0, y, x0+CELL-1, y+CELL-1], fill=(255,255,255) if bit else (0,0,0))
        x += nbits * CELL + CELL

def read_control(img: Image.Image) -> Dict[str,int]:
    x = 4; y = CTRL_Y0 + 4
    out: Dict[str,int] = {}
    for name, nbits in FIELDS:
        val = 0
        for b in range(nbits):
            x0 = x + b * CELL
            roi = img.crop((x0, y, x0+CELL, y+CELL))
            bit = 1 if ImageStat.Stat(roi).mean[0] > 127 else 0
            val = (val<<1) | bit
        out[name] = val
        x += nbits * CELL + CELL
    return out

@dataclass
class ScreenState:
    pc: int = 0
    mailbox_out: int = 0

class Screen(Protocol):
    id: int
    def draw(self, state: ScreenState, px) -> None: ...
    def decode(self, img: Image.Image) -> Dict[str,int]: ...
    def update(self, decoded: Dict[str,int], state: ScreenState, inputs: Dict) -> Tuple[ScreenState, int]: ...
    def accept(self, decoded: Dict[str,int]) -> bool: ...
```

from typing import Dict
from py2px import Px
from screen_base import Screen, ScreenState
from screen_hello import HelloScreen
from screen_menu import MenuScreen

REGISTRY: Dict[int, Screen] = {
    HelloScreen.id: HelloScreen(),
    MenuScreen.id: MenuScreen(),
}

def read_inputs() -> Dict:
    # stub: inject input events here (keyboard hook, queue, or scripted)
    return {"key_pressed": False, "key_q": False}

def run_once(width=320, height=200, frames=120, out_dir="out_screens"):
    px = Px(width, height, 60, out_dir)
    screen: Screen = REGISTRY[1]
    state = ScreenState()

    for _ in range(frames):
        # 1) draw current screen
        screen.draw(state, px)
        px.tick()

        # 2) capture & decode what we actually drew
        img = px._img  # digital path: ground-truth framebuffer
        decoded = screen.decode(img)

        # 3) acceptance for this screen
        assert screen.accept(decoded), f"Screen {screen.id} failed acceptance: {decoded}"

        # 4) update state + possibly switch screen
        state, next_id = screen.update(decoded, state, read_inputs())
        screen = REGISTRY.get(next_id, screen)

if __name__ == "__main__":
    run_once()
```

## Why this works well “one screen at a time”

* **Tight scope**: you finish and lock one screen (draw/decoded/accepted) before adding the next.

* **Same loop later**: swap capture for a camera and replace `read_control` with the real fiducial/ECC decoder.
* **No hidden state**: everything needed to advance is on the screen (via the control strip), so self‑hosting is feasible.

Love that model. Treat the infinite map as a **sparse, chunked canvas** and each “program” as a **stack** (frames/keyframes) anchored to a rectangular region. Then the runtime just pages in whatever stacks intersect the current viewport and advances only those.

Here’s a clean, workable plan (and a tiny skeleton) to make that real.

# How we lay the world out

* **Global coords:** 2‑D integer plane (…,−2,−1,0,1,2,…).
* **Chunking:** split space into fixed chunks, e.g. 256×256 px. This keeps paging/rendering O(visible).
* **Program = Stack:** a file with:

  * `program_id` (content‑hash),
  * `bbox` (x0,y0,x1,y1) where it lives on the map,
  * `frames` (list of immutable frames or keyframes + deltas),
  * `state` (PC, mailboxes…),
  * `mount_at` (top‑left coordinate the stack is anchored to).
* **Multiple programs per area:** Any chunk can hold N stacks; we z‑order by `layer` or `z`.
* **Files on the map:** The map *is* a filesystem. Think paths like `/map/{chunk_x}/{chunk_y}/index.json` listing the stacks inside that chunk; the stack files live at `/programs/{program_id}.pxstack`.

# How the loop works (high level)

1. **Viewport → chunks:** compute which chunks overlap the viewport.
2. **Query stacks:** for those chunks, list stacks whose `bbox` intersects the viewport.
3. **Advance only visible stacks:** for each stack, run `read → decide → draw` for *its next frame*.
4. **Composite:** render stacks into the framebuffer by z‑order.
5. **Persist deltas:** if the stack advanced, write back its new `state` and append frame refs (content‑addressed).

# Addressing + placement

* **Drop a program** at (x,y) by writing a stack header:

  ```json
  {
    "program_id": "sha256:…",
    "mount_at": {"x": 12345, "y": -222},
    "bbox": {"x0":12345,"y0":-222,"x1":12345+W,"y1":-222+H},
    "z": 10,
    "frames": ["blake3:…", "blake3:…"],
    "state": {"pc":0,"mailbox_out":0}
  }
  ```
* **Moving/resize** = update `mount_at`/`bbox` (and reindex chunk membership for edges).

# Storage choices (simple & robust)

* **Chunk index** (append‑only JSON): `/map/{cx}/{cy}/index.json` → array of `{stack_id, bbox, z}`.
* **Stack file**: `/stacks/{stack_id}.json` → header + pointers to frame blobs.
* **Frames**: `/frames/{hash}.png` or `.pxsheet` (tile+ECC format).
* **Content‑address all blobs** for dedup across programs.

# Minimal runner skeleton (Python)

```python
# map_runtime.py
from dataclasses import dataclass
from typing import List, Dict, Tuple, Iterable, Optional
from PIL import Image

CHUNK = 256

def chunk_of(x:int, y:int) -> Tuple[int,int]:
    return x // CHUNK if x>=0 else -((-x-1)//CHUNK)-1, y // CHUNK if y>=0 else -((-y-1)//CHUNK)-1

@dataclass
class BBox: x0:int; y0:int; x1:int; y1:int
@dataclass
class StackHeader:
    stack_id: str
    program_id: str
    bbox: BBox
    z: int
    state: Dict
    frames: List[str]  # content hashes for frames or keyframes

class ChunkIndex:
    def __init__(self, store): self.store = store  # simple KV (path->bytes)
    def list_stacks(self, cx:int, cy:int) -> List[str]:
        data = self.store.get_json(f"/map/{cx}/{cy}/index.json") or []
        return [e["stack_id"] for e in data]
    def upsert(self, cx:int, cy:int, entry:Dict): ...
    def remove(self, cx:int, cy:int, stack_id:str): ...

class StackStore:
    def __init__(self, store): self.store = store
    def load(self, stack_id:str) -> StackHeader:
        d = self.store.get_json(f"/stacks/{stack_id}.json")
        bb = BBox(**d["bbox"])
        return StackHeader(stack_id=d["stack_id"], program_id=d["program_id"], bbox=bb, z=d["z"], state=d["state"], frames=d["frames"])
    def save(self, hdr:StackHeader): ...

class FrameStore:
    def __init__(self, store): self.store = store
    def get_frame(self, h:str) -> Image.Image: return self.store.get_image(f"/frames/{h}.png")
    def put_frame(self, img:Image.Image) -> str:
        # hash->path, write if missing, return hash
        ...

class MapRuntime:
    def __init__(self, store, renderer, decoder, reducer):
        self.chunks = ChunkIndex(store)
        self.stacks = StackStore(store)
        self.frames = FrameStore(store)
        self.renderer = renderer  # program_id -> draw(state) -> PIL.Image
        self.decoder = decoder    # program_id -> decode(image) -> dict
        self.reducer = reducer    # program_id -> update(decoded,state,inputs) -> new_state

    def stacks_in_view(self, view:BBox) -> List[StackHeader]:
        cxs = range(chunk_of(view.x0,0)[0], chunk_of(view.x1-1,0)[0]+1)
        cys = range(chunk_of(0,view.y0)[1], chunk_of(0,view.y1-1)[1]+1)
        seen, out = set(), []
        for cx in cxs:
            for cy in cys:
                for sid in self.chunks.list_stacks(cx, cy):
                    if sid in seen: continue
                    seen.add(sid)
                    hdr = self.stacks.load(sid)
                    if not (hdr.bbox.x1 <= view.x0 or hdr.bbox.x0 >= view.x1 or hdr.bbox.y1 <= view.y0 or hdr.bbox.y0 >= view.y1):
                        out.append(hdr)
        return sorted(out, key=lambda h: h.z)

    def step_visible(self, view:BBox, inputs:Dict) -> Image.Image:
        # 1) render + advance each visible stack
        layers: List[Tuple[int, Image.Image, StackHeader]] = []
        for hdr in self.stacks_in_view(view):
            img = self.renderer.draw(hdr.program_id, hdr.state, hdr.bbox)  # full stack frame (local coords)
            decoded = self.decoder.decode(hdr.program_id, img)
            new_state = self.reducer.update(hdr.program_id, decoded, hdr.state, inputs)
            if new_state != hdr.state:
                hdr.state = new_state
                # optional: persist frame (content addressed)
                # h = self.frames.put_frame(img); hdr.frames.append(h); self.stacks.save(hdr)
            layers.append((hdr.z, img, hdr))
        # 2) composite in z-order into final framebuffer
        fb = Image.new("RGB", (view.x1-view.x0, view.y1-view.y0), "#000000")
        for _, img, hdr in layers:
            ox, oy = hdr.bbox.x0 - view.x0, hdr.bbox.y0 - view.y0
            fb.paste(img, (ox, oy), mask=None)
        return fb
```

# How you “put different files in different places”

* **Mounting:** `place_program(program_id, at=(x,y), z=…)` creates a `stack_id`, sets `bbox` from the program’s declared size, and writes it into all affected **chunk indices** (edges/corners too).
* **Multiple versions:** You can mount *the same* `program_id` many times (different `stack_id`s, states, positions).
* **Moving:** remove from old chunk indices, update `bbox`, re‑index in new chunks.
* **Deleting:** remove the `stack_id` from indices (the content‑addressed frames remain if referenced elsewhere).

# How “big stack = complex program” works

* The **stack’s length** (or keyframe count) grows as it runs. We never rewrite frames; we append.
* **Code = data on the map:** if a program compiles down to frame‑sheets (`.pxsheet`), the stack is literally its sheets.
* A tiny control strip on each frame carries `{PROGRAM_ID, STACK_ID, PC, CRC}`, so any future capture can **self‑locate** and resume.

# Nice extras (when you want them)

* **Quadtree index** for faster spatial queries (optional; chunking is usually enough).
* **Transactional writes** for stack/state updates (append‑only journal → compact).
* **Content addressing everywhere** (BLAKE3/xxh3) so identical frames across programs don’t duplicate storage.
* **Cross‑stack messaging** via a small, map‑scoped mailbox: `/bus/{x_chunk}_{y_chunk}.q` for local broadcasts.
* **Permissions**: a stack can declare “collides\_with” policy (e.g., top‑most wins, blend, or occlude).
