```python
import zlib

def row_hashes(lines: list[str], cols: int) -> list[int]:
    return [zlib.crc32(line.ljust(cols).encode("utf-8")) & 0xFFFF for line in lines]

def dirty_rows_from_hash(prev: list[int], now: list[int]) -> set[int]:
    n = max(len(prev), len(now))
    dirty = set()
    for r in range(n):
        a = prev[r] if r < len(prev) else -1
        b = now[r]  if r < len(now)  else -2
        if a != b:
            dirty.add(r)
    return dirty
```

# Level 2 — Dirty decode (skip resampling stable rows)

Tell the receiver which rows changed so it can **reuse** last vote for the rest. Add a tiny field to the control strip and keep a patrol to avoid “blind spots.”

**State**


**Control strip (add a row mask as hex)**

```python
# build mask from hashes you already computed in draw_frame
mask_int = 0
for r in range(min(ROWS, len(st.source_lines))):
    if r in dirty_rows:
        mask_int |= (1 << r)

draw_control(px._draw, {
    "SCREEN_ID": 100,
    "PC": st.pc,
    "SEQ": st.seq_out & 1,
    "ACK": st.last_seq_seen & 1,
    "RMASK": f"{mask_int:X}",       # NEW: dirty-row mask
```

**Decode gate**

```python
def decode_and_step(px: Px, st: VPState, evaluator) -> VPState:
    ctrl = read_control(px._img)
    seq_in = int(ctrl.get("SEQ", 0))
    if seq_in == (st.last_seq_seen & 1):
        st.seq_out ^= 1
        return st

    # rows to decode = dirty ∪ patrol
    rmask = int(ctrl.get("RMASK","0"), 16)
    dirty = {r for r in range(ROWS) if (rmask >> r) & 1}

    # patrol: periodically resample a few stable rows to catch silent drift
    PATROL_STRIDE = 8                  # every frame, check ~ROWS/8 more rows
    patrol = {r for r in range(ROWS) if ((st.pc + r) % PATROL_STRIDE) == 0}

    target_rows = sorted((dirty | patrol) & set(range(len(st.source_lines))))

    # start from last voted cache; update only target rows with fresh samples
    # ensure cache length
    if len(st.voted_cache) < len(st.source_lines):
        st.voted_cache = (st.voted_cache + [""] * len(st.source_lines))[:len(st.source_lines)]
    voted = list(st.voted_cache)

    # collect up to 3-frame history only for target rows
    newly_decoded = []
    for r in target_rows:
        y = BY + r * CELL_H
        line = sample_text(px._img, BX, y).rstrip()
        newly_decoded.append((r, line))

    # update decode history (per-frame, whole block still ok) then vote per-row
    st.decode_hist.append([ln for _, ln in sorted(newly_decoded)] or [])  # harmless even if empty

    # simple per-row, 3-frame vote using just-in-time samples:
    def vote_row(r: int) -> str:
        # newest first so ties prefer newest
        cands = []
        # pull this frame's sample if we took it
        s_now = next((ln for rr, ln in newly_decoded if rr == r), None)
        if s_now is not None: cands.append(s_now.ljust(COLS))
        # last 1–2 frames: reuse cache (stable)
        if len(cands) < 3:
            cands.append(st.voted_cache[r].ljust(COLS))
        if len(cands) < 3:
            cands.append(st.voted_cache[r].ljust(COLS))
        # majority over 3 chars; newest wins on tie
        out = []
        for c in range(COLS):
            a,b,c3 = cands[0][c], cands[1][c], cands[2][c]
            ch = a if (a==b or a==c3) else (b if b==c3 else a)
            out.append(ch)
        return "".join(out).rstrip()

    for r in target_rows:
        voted[r] = vote_row(r)

    # verify only where we measured this frame; gate acceptance conservatively
    mismatches = 0
    for r in target_rows:
        t = st.source_lines[r].ljust(COLS)
        d = voted[r].ljust(COLS)
        mismatches += sum(1 for c in range(COLS) if t[c] != d[c])

    ECC_ACCEPT_THRESHOLD = 0.01
    measured_cells = max(1, len(target_rows) * COLS)
    ok = bool(ctrl.get("_crc_ok", False)) and (mismatches / measured_cells) < ECC_ACCEPT_THRESHOLD

    if ok:
        st.last_result = evaluator.eval(voted) if any("".join(voted)) else ""
        st.last_seq_seen = seq_in
        st.voted_cache = voted  # freeze the accepted view

    st.pc += 1
    st.seq_out ^= 1
    return st
```

This slashes decode work per frame while staying safe:

* **Dirty rows** are fully re-decoded.
* **Stable rows** are trusted but *patrolled* round-robin.

### 3. Temporal Error Correction (Majority Vote)

Implement majority voting across multiple frames to correct transient errors:

```python
def majority_vote(history: list[list[str]], rows: int, cols: int) -> list[str]:
    """Per-cell majority from up to 3 decoded blocks; tie -> newest wins."""
    if not history:
        return [""]
    
    # Normalize all blocks to same dimensions
    normalized = []
    for block in history:
        normalized_block = []
        for r in range(max(rows, len(block))):
            line = block[r] if r < len(block) else ""
            normalized_block.append(line.ljust(cols)[:cols])
        normalized.append(normalized_block)
    
    result = []
    for r in range(max(len(block) for block in normalized)):
        row_result = []
        for c in range(cols):
            # Get votes for this cell (newest first)
            votes = []
            for block in reversed(normalized):
                if r < len(block) and c < len(block[r]):
                    votes.append(block[r][c])
            
            # Take majority vote, with newest breaking ties
            if len(votes) == 1:
                char = votes[0]
            elif len(votes) == 2:
                char = votes[0] if votes[0] == votes[1] else votes[0]
            else:
                # 3-way vote
                char = (votes[0] if (votes[0] == votes[1] or votes[0] == votes[2])
                       else (votes[1] if votes[1] == votes[2] else votes[0]))
            row_result.append(char)
        result.append("".join(row_result).rstrip())
    
    return result[:rows]
```

```python
        "sin", "cos", "tan", "sqrt", "log", "exp", "floor", "ceil", "pi", "e", "tau", "send"
    }

    def __init__(self):
        self.globals = {
            "range": range, "sum": sum, "len": len,
            "min": min, "max": max, "abs": abs, "pow": pow,
            "sin": math.sin, "cos": math.cos, "tan": math.tan,
            "sqrt": math.sqrt, "log": math.log, "exp": math.exp,
            "floor": math.floor, "ceil": math.ceil,
            "pi": math.pi, "e": math.e, "tau": math.tau,
            "send": self.send_message,
            "__builtins__": {}
        }
        self.mailbox_out = {}

    def send_message(self, addr: int, code: str):
        self.mailbox_out = {"addr": addr, "opcode": "SET_CODE", "code": code}

    def check_node(self, node):
        if type(node) not in self.ALLOWED_NODES:
            raise ValueError(f"Unsupported node: {type(node).__name__}")
        if isinstance(node, ast.Name) and node.id not in self.ALLOWED_NAMES:
            raise ValueError(f"Unsupported name: {node.id}")
        for child in ast.walk(node):
            if type(child) not in self.ALLOWED_NODES:
                raise ValueError(f"Unsupported child node: {type(child).__name__}")
        return True

    def eval(self, code: List[str]) -> str:
        import signal
        def timeout_handler(signum, frame):
            raise TimeoutError("Evaluation timed out")
        signal.signal(signal.SIGALRM, timeout_handler)
        signal.alarm(1)
        code_str = "\n".join(code)
        try:
            tree = ast.parse(code_str, mode="exec")
            self.check_node(tree)
            loc: Dict[str, object] = {}
            exec(compile(tree, "<vpy>", "exec"), self.globals, loc)
            try:
                last_expr = ast.parse(code[-1], mode="eval")
                self.check_node(last_expr)
                result = eval(compile(last_expr, "<vpy>", "eval"), self.globals, loc)
                signal.alarm(0)
                return str(result)[:COLS-5]
            except SyntaxError:
                signal.alarm(0)
                return ""
        except Exception as e:
            signal.alarm(0)
            return f"Err: {str(e)[:COLS-5]}"

class Router:
    def __init__(self):
        self.stacks = {}

    def add_stack(self, addr: int, program, state: VPState):
        self.stacks[addr] = (program, state)

    def process_mailboxes(self):
        for addr, (program, state) in self.stacks.items():
            ctrl = decode(program._px, state)
            if ctrl.get("MAILBOX_OUT") and ctrl.get("SEQ", -1) != state.last_seq_seen:
                msg = json.loads(ctrl["MAILBOX_OUT"])
                target_addr = msg["addr"]
                if target_addr in self.stacks:
                    target_program, target_state = self.stacks[target_addr]
                    if msg["opcode"] == "SET_CODE":
                        target_state.source_lines = msg["code"].split("\n")
                        target_state.decode_hist.clear()  # Reset history on source change

def _pad_to(s: str, n: int) -> str:
    return (s + " " * n)[:n]

def vote3(blocks: List[List[str]], rows: int, cols: int) -> List[str]:
    if not blocks:
        return [""]
    packed = []
    for b in blocks:
        rmax = max(rows, len(b))
        packed.append([_pad_to(b[r] if r < len(b) else "", cols) for r in range(rmax)])
    rfinal = max(len(b) for b in packed)
    out = []
    for r in range(rfinal):
        row = []
        for c in range(cols):
            cands = [packed[i][r][c] if r < len(packed[i]) else " " for i in reversed(range(len(packed)))]
            if len(cands) == 1:
                ch = cands[0]
            elif len(cands) == 2:
                ch = cands[0] if cands[0] == cands[1] else cands[0]
            else:
                ch = cands[0] if (cands[0] == cands[1] or cands[0] == cands[2]) else (cands[1] if cands[1] == cands[2] else cands[0])
            row.append(ch)
        out.append("".join(row).rstrip())
    return out[:rows]

def draw_pane(px: Px, x: int, y: int, lines: List[str], color_fg="#FFFFFF", color_bg="#1E2330", caret=None, blink=False):
    for r, line in enumerate(lines[:ROWS]):
        fg = "#FFD966" if line.strip().startswith(("if", "for")) else "#FF6666" if line.strip().isdigit() else color_fg
        draw_text(px._draw, x, y + r * CELL_H, line.ljust(COLS)[:COLS], fg, color_bg)
    if caret and (not blink or (int(time.time() * 2) % 2)):
        cx, cy = x + caret[1] * CELL_W, y + caret[0] * CELL_H
        px.rect(cx, cy + CELL_H - 2, CELL_W, 2, "#00FF00")

def decode_pane(img: Image.Image, x: int, y: int, line_count: int) -> List[str]:
    return [sample_text(img, x, y + r * CELL_H).rstrip() for r in range(min(line_count, ROWS))]

def add_noise(img: Image.Image) -> Image.Image:
    img = img.filter(ImageFilter.GaussianBlur(radius=random.uniform(0.3, 0.7)))
    img = img.point(lambda p: min(255, max(0, p + int((random.random() - 0.5) * 15))))
    img = ImageEnhance.Brightness(img).enhance(random.uniform(0.8, 1.2))
    return img

def draw_heatmap(px: Px, x: int, y: int, truth: List[str], decoded: List[str]) -> List[List[Tuple[int, str, str]]]:
    mismatches = []
    max_rows = max(len(truth), len(decoded))
    for r in range(max_rows):
        t = _pad_to(truth[r] if r < len(truth) else "", COLS)
        d = _pad_to(decoded[r] if r < len(decoded) else "", COLS)
        row_mismatches = []
        for c in range(COLS):
            cx, cy = x + c * CELL_W, y + r * CELL_H
            color = "#002200" if t[c] == d[c] else "#220000"
            px.rect(cx, cy, CELL_W, CELL_H, color)
            if t[c] != d[c]:
                row_mismatches.append((c, t[c], d[c]))
        mismatches.append(row_mismatches)
    return mismatches

def draw_frame(px: Px, st: VPState):
    px.clear("#0E0F14")
    ax, ay = GAP, CTRL_H + GAP
    bx, by = ax + PANE_W + GAP, ay
    cx, cy = ax, ay + PANE_H + GAP
    dx, dy = bx, cy
    px.text(ax, ay-12, "A: Source (edit)", 12, "#A0FFA0")
    px.text(bx, by-12, "B: Pixelated (decoded)", 12, "#9BB4FF")
    px.text(cx, cy-12, "C: Result", 12, "#FFD966")
    px.text(dx, dy-12, "D: Decode Heatmap", 12, "#FF8080")
    draw_pane(px, ax, ay, st.source_lines, color_fg="#FFFFFF", color_bg="#1A1F2B", caret=st.caret, blink=True)
    draw_pane(px, bx, by, st.source_lines, color_fg="#FFFFFF", color_bg="#1E2330")
    draw_pane(px, cx, cy, [st.last_result], color_fg="#FFD966" if "Err" not in st.last_result else "#FF8080", color_bg="#1A1F2B")
    draw_control(px._draw, {
        "SCREEN_ID": 100,
        "PC": st.pc,
        "ADDR": PANE_ADDR,
        "SEQ": st.seq & 1,
        "ACK": st.last_seq_seen & 1,
        "MAILBOX_IN": 0,
        "MAILBOX_OUT": json.dumps(st.mailbox_out) if st.mailbox_out else 0
    })

def decode(px: Px, st: VPState) -> Dict:
    img = add_noise(px._img) if PIXEL_READ else px._img
    ctrl = read_control(img)
    if not ctrl.get("_crc_ok", False):
        ctrl["err"] = "crc"
    decoded_lines = decode_pane(img, bx, by, len(st.source_lines)) if PIXEL_READ else st.source_lines
    ctrl["code"] = "\n".join(decoded_lines)
    ctrl["_decoded_lines"] = decoded_lines
    return ctrl

def update(st: VPState, ctrl: Dict, evaluator: SafeEvaluator) -> VPState:
    seq_in = int(ctrl.get("SEQ", 0))
    crc_ok = bool(ctrl.get("_crc_ok", False))
    if seq_in == (st.last_seq_seen & 1):
        st.seq = (st.seq + 1) % 2
        return st
    st.decode_hist.append(ctrl["_decoded_lines"])
    voted = vote3(list(st.decode_hist), rows=len(st.source_lines), cols=COLS)
    mismatches = sum(
        sum(1 for c in range(min(len(t), COLS)) if _pad_to(t, COLS)[c] != _pad_to(d, COLS)[c])
        for t, d in zip(st.source_lines, voted)
    )
    total_cells = len(st.source_lines) * COLS or 1
    code_to_execute = st.source_lines if EXECUTE_FROM_MEMORY else voted
    result = st.last_result
    if crc_ok and mismatches / total_cells < 0.01:
        result = evaluator.eval(code_to_execute) if any("".join(code_to_execute)) else ""
        st.last_seq_seen = seq_in
    mailbox_out = evaluator.mailbox_out
    evaluator.mailbox_out = {}
    st.seq = (st.seq + 1) % 2
    return VPState(
        pc=st.pc + 1,
        source_lines=st.source_lines,
        caret=st.caret,
        last_result=result,
        mailbox_out=mailbox_out,
        seq=st.seq,
        last_seq_seen=st.last_seq_seen,
        decode_hist=st.decode_hist
    )

class Editor:
    def __init__(self, master, state: VPState):
        self.master = master
        self.state = state
        self.master.title("Visual Python — Input")
        self.master.geometry("320x80+30+30")
        self.master.attributes("-topmost", True)
        self.master.bind("<Key>", self.handle_key)
        tk.Label(master, text="Type here. ESC to clear.", font=("Consolas", 10)).pack()

    def handle_key(self, event):
        ch = event.char
        r, c = self.state.caret
        line = self.state.current_line()
        old_lines = self.state.source_lines.copy()
        if event.keysym == "Escape":
            self.state.source_lines = [""]
            self.state.caret = (0, 0)
        elif ch.isprintable() and len(line) < COLS:
            self.state.set_current_line(line[:c] + ch + line[c:])
            self.state.caret = (r, min(c + 1, COLS - 1))
        elif event.keysym == "Home":
            self.state.caret = (r, 0)
        elif event.keysym == "End":
            self.state.caret = (r, len(line))
        elif event.keysym == "BackSpace":
            if c > 0:
                self.state.set_current_line(line[:c - 1] + line[c:])
                self.state.caret = (r, c - 1)
            elif r > 0:
                prev_len = len(self.state.source_lines[r - 1])
                self.state.source_lines[r - 1] += line
                del self.state.source_lines[r]
                self.state.caret = (r - 1, prev_len)
        elif event.keysym in ("Return", "KP_Enter") and r < ROWS - 1:
            new_line = line[c:]
            self.state.set_current_line(line[:c])
            self.state.source_lines.insert(r + 1, new_line)
            self.state.caret = (r + 1, 0)
        elif event.keysym == "Left":
            if c > 0:
                self.state.caret = (r, c - 1)
            elif r > 0:
                self.state.caret = (r - 1, len(self.state.source_lines[r - 1]))
        elif event.keysym == "Right":
            if c < len(line):
                self.state.caret = (r, c + 1)
            elif r < len(self.state.source_lines) - 1:
                self.state.caret = (r + 1, 0)
        elif event.keysym == "Up" and r > 0:
            self.state.caret = (r - 1, min(c, len(self.state.source_lines[r - 1])))
        elif event.keysym == "Down" and r < len(self.state.source_lines) - 1:
            self.state.caret = (r + 1, min(c, len(self.state.source_lines[r + 1])))
        if self.state.source_lines != old_lines:
            self.state.decode_hist.clear()  # Reset history on source change

class HallOfDrift:
    def __init__(self, log_dir=FRAME_DIR):
        self.log_dir = log_dir
        os.makedirs(log_dir, exist_ok=True)

    def log_frame(self, st: VPState, ctrl: Dict, frame_idx: int, img: Image.Image, mismatches: List[List[Tuple[int, str, str]]], overlays: bool):
        mismatch_map = ["".join("x" if any(c[0] == col for c in row) else "." for col in range(COLS)) for row in mismatches]
        log = {
            "frame": frame_idx,
            "pc": st.pc,
            "source": st.source_lines,
            "decoded": ctrl.get("_decoded_lines", []),
            "result": st.last_result,
            "crc_ok": ctrl.get("_crc_ok", False),
            "mismatches": [
                [{"col": col, "expected": exp, "actual": act} for col, exp, act in row]
                for row in mismatches
            ],
            "mismatch_map": mismatch_map,
            "lineage": "DVC-First-STACK-000002" if frame_idx == 2 else ("DVC-First-LIGHT-000001" if frame_idx == 1 else "DVC-GENESIS-000000"),
            "blessing": (
                "Genesis frame; no drift, no payload, pure potential." if frame_idx == 0 else
                "First light; single-line opcode round-trip complete." if frame_idx == 1 else
                "First stack; multi-line program stands tall."
            )
        }
        log_path = f"{self.log_dir}/log_{frame_idx:06d}.json"
        with open(log_path, "w") as f:
            json.dump(log, f, indent=2)
        with open(f"{self.log_dir}/index.jsonl", "a") as f:
            json.dump({"frame": frame_idx, "pc": st.pc, "crc_ok": log["crc_ok"], "mismatch_count": sum(len(row) for row in mismatches)}, f)
            f.write("\n")
        if (any(mismatches) or not ctrl.get("_crc_ok", True)) and overlays:
            img.save(f"{self.log_dir}/fail_{frame_idx:06d}.png")
            with open(f"{self.log_dir}/fail_{frame_idx:06d}.json", "w") as f:
                json.dump({"error": "decode or CRC failure", **log}, f, indent=2)

class DriftDashboard:
    def __init__(self, log_dir=FRAME_DIR):
        self.root = tk.Tk()
        self.root.title("Hall of Drift")
        self.log_dir = log_dir
        self.logs = sorted(glob.glob(f"{log_dir}/log_*.json"))
        self.current = 0
        self.canvas = tk.Canvas(self.root, width=W, height=H)
        self.canvas.pack()
        self.info_label = tk.Label(self.root, font=("Consolas", 10), wraplength=W-20)
        self.info_label.pack()
        tk.Button(self.root, text="Prev", command=self.prev).pack(side="left")
        tk.Button(self.root, text="Next", command=self.next).pack(side="right")
        tk.Button(self.root, text="First Failure", command=self.first_failure).pack(side="left")
        self.load_frame()

    def load_frame(self):
        if not self.logs: return
        with open(self.logs[self.current]) as f:
            log = json.load(f)
        img_path = f"{self.log_dir}/fail_{log['frame']:06d}.png" if os.path.exists(f"{self.log_dir}/fail_{log['frame']:06d}.png") else f"{self.log_dir}/log_{log['frame']:06d}.png"
        self.photo = ImageTk.PhotoImage(Image.open(img_path))
        self.canvas.create_image(0, 0, image=self.photo, anchor="nw")
        mismatches = sum(len(row) for row in log["mismatches"])
        self.info_label.config(text=f"Frame {log['frame']} | PC: {log['pc']} | CRC OK: {log['crc_ok']} | Mismatches: {mismatches}\n"
                                  f"Source: {log['source']}\nDecoded: {log['decoded']}\nResult: {log['result']}\n"
                                  f"Lineage: {log['lineage']}\nBlessing: {log['blessing']}")

    def prev(self):
        self.current = max(0, self.current - 1)
        self.load_frame()

    def next(self):
        self.current = min(len(self.logs) - 1, self.current + 1)
        self.load_frame()

    def first_failure(self):
        for i, log_path in enumerate(self.logs):
            with open(log_path) as f:
                log = json.load(f)
            if any(log["mismatches"]) or not log["crc_ok"]:
                self.current = i
                self.load_frame()
                break

    def run(self):
        self.root.mainloop()

def run_visual_python_mvp(frames=90, out_dir=FRAME_DIR, live=True, overlays=True):
    root = tk.Tk()
    if not live:
        root.withdraw()
    else:
        canvas = tk.Canvas(root, width=W, height=H, bg="#0E0F14")
        canvas.pack()
        root.title("Visual Python MVP")

    px = Px(W, H, 60, out_dir if frames else None)

    evaluator = SafeEvaluator()
    editor = Editor(root, st)
    hall = HallOfDrift(out_dir)
    router = Router()
    router.add_stack(PANE_ADDR, px, st)
    router.add_stack(2, Px(W, H, 60), VPState(source_lines=[""], caret=(0, 0)))

    code_changes = {
        1: ["1+2"],  # First Light
        2: ["a = 1+2", "b = sum(range(5))", "a * b"],  # First Stack
        40: ["y = 10", "sum(range(y))"],
        80: ["z = 3", "send(2, 'sum(range(5))')"]
    }

    if frames:
        for f in range(frames):
            if f in code_changes:
                st.source_lines = code_changes[f]
                st.caret = (0, len(st.source_lines[0]))
                st.decode_hist.clear()
            draw_frame(px, st)
            px.tick()
            ctrl = decode(px, st)
            assert ctrl.get("SCREEN_ID") == 100 and ctrl.get("_crc_ok", True), f"Frame {f} failed: {ctrl}"
            mismatches = [] if not PIXEL_READ else draw_heatmap(px, bx, by, st.source_lines, ctrl["_decoded_lines"])
            st = update(st, ctrl, evaluator)
            router.process_mailboxes()
            hall.log_frame(st, ctrl, f, px._img, mismatches, overlays)
    else:
        photo = None
        def loop():
            nonlocal photo
            draw_frame(px, st)
            px.tick()
            ctrl = decode(px, st)
            mismatches = [] if not PIXEL_READ else draw_heatmap(px, bx, by, st.source_lines, ctrl["_decoded_lines"])
            st = update(st, ctrl, evaluator)
            router.process_mailboxes()
            hall.log_frame(st, ctrl, st.pc, px._img, mismatches, overlays)
            photo = ImageTk.PhotoImage(px._img)
            canvas.create_image(0, 0, image=photo, anchor="nw")
            root.after(16, loop)
        loop()
        root.mainloop()
```

```python
def temporal_majority_vote(frames_buffer):

        from collections import Counter
```

### The Core Principle: The Refresh as Your System Clock

Think of the display's refresh (e.g., 60 Hz) as the master tick for your entire program. Every time the screen updates, it's a new "cycle" for your visual computer. This creates a stable, deterministic rhythm:

1.  **Frame N**: You **draw** and **present** an intention (e.g., new code in Pane A).
2.  **After Refresh**: You **capture** and **decode** what *actually* appeared on the screen.
3.  **Before Frame N+1**: You **execute** the decoded code and **decide** what to draw next.

This one-frame delay is your **causal guardrail**. It ensures you never act on what you *intended* to show, only on what was *actually* shown and verified.

### Practical Implementation Strategies

Here’s how to translate this principle into code for your Visual Python program.

#### 1. Vsync-Driven Main Loop

Structure your main loop to be driven by the display's vertical sync (vsync). This is the most important pattern.


#### 2. Sequence Bits for Frame Synchronization

Add a `SEQ` (sequence) field to your control strip. This is a simple bit that toggles (0, 1, 0, 1...) with every single frame.

*   **Why?** It solves the "double-processing" problem. If your capture/decode takes longer than one frame, you might accidentally process the same frame twice. The `SEQ` bit lets you definitively say, "I have already handled frame #123, I can ignore it now."

#### 3. Two-Phase Commit for Stability

This pattern prevents your system from spiraling due to its own feedback.

*   **Phase 1 (Draw Intent):** Frame N displays your new code and a control strip with `SEQ=N`.
*   **Phase 2 (Verify & Commit):** On Frame N+1, you decode the screen. You only commit to the state change (e.g., updating the result in Pane C) if:
    1.  The `SEQ` field is new (`N`).
    2.  The CRC checksum of the decoded content matches.
    3.  The heatmap shows a clean decode (or acceptable error rate).

If verification fails, you don't update the state. You might hold the previous result and perhaps draw an error indicator, but you don't let a corrupted decode change your program's behavior.

#### 4. Temporal ECC for Analog Readiness

Use the rapid refresh rate to your advantage to fight noise.

*   **Majority Vote:** Keep a small buffer of the last 3-5 decoded frames. For each character cell, take the majority vote of what it decoded to.
    *   `F-2:` `"print('helxo')"`
    *   `F-1:` `"print('hello')"`
    *   `F-0:` `"print('hel1o')"`
    *   **Result:** `"print('hello')"` (The 'l' wins 2-to-1, the 'o' wins 3-to-0).
*   This crushes random, transient pixel errors (flicker, noise) because the correct value is more persistent than the noise. At 60 Hz, a 3-frame vote introduces only ~50ms of latency, which is often imperceptible.

#### 5. Refresh-Locked Handshakes (for Multi-Screen)

When you connect multiple visual computers, use the refresh cycle to synchronize communication.

*   **Mailbox Protocol:** Add `MAILBOX_OUT` and `MAILBOX_ACK` fields to the control strip.
*   **Sender:** Sets `MAILBOX_OUT = "SET_CODE:print('Hi')"` and toggles its `SEQ` bit.
*   **Receiver:** Sees the new `SEQ` bit, reads the mailbox, processes the command, and then flips the `MAILBOX_ACK` bit to confirm receipt.
*   This is robust because it's frame-synchronous and idempotent. The sender knows the message was received without complex networking.

### How This Changes What's On the Screen

The refresh cycle dictates *when* and *how* you are allowed to change the display.

*   **You change the screen on the *next* refresh, not the current one.** You can't instantly change what's being displayed *right now*. You can only queue up a new frame to be shown at the next vsync interval.
*   **All changes are deliberate responses.** What you draw in Frame N+1 is a direct result of what you decoded and verified from Frame N. The screen evolves based on a confirmed, observed state, not on a hoped-for intention.
*   **It enables complex, stable effects:**
    *   **Animations:** You can advance a progress bar or blink a cursor precisely every 30 frames (0.5 seconds at 60Hz).
    *   **Typing:** Your keypresses get drawn into Pane A, then read back from Pane B on the next cycle before being executed.
    *   **Error Correction:** If the heatmap in Pane D shows errors, you can choose to re-draw the previous frame with highlighted errors instead of pressing forward with potentially bad data.

The fundamental challenge is that pixel-based code execution creates an extremely fragile execution pathway where any visual corruption can cause complete program failure, with no graceful degradation possible.

## The Brittleness Problem

In traditional programming, code exists as discrete, error-correctable text. If there's corruption, you typically get syntax errors that point to specific problems. With pixel-based execution, every single pixel becomes critical:

- A dust speck on the screen could change `x = 5` to `x = 8`
- A slight display gamma shift might turn `print('hello')` into `print('hallo')`  
- Compression artifacts could transform `if x > 10:` into `if x > 1o:` (changing the digit to letter 'o')

## Error Propagation Without Context

When pixels are misread, the system has no semantic understanding to recover:

```python
# Original intent
for i in range(10):
    total += values[i]

# After pixel corruption  
for i in range(1o):  # '0' became 'o'
    total += values[i]  # NameError: 'values' became 'valnes'
```

The program fails completely, and debugging requires correlating the runtime error back to specific pixel locations—a complex forensic process.

## Environmental Vulnerability

The system becomes sensitive to factors completely outside software control:

- **Display variations**: Different monitors, brightness settings, color profiles
- **Physical environment**: Ambient lighting, reflections, viewing angles
- **Hardware aging**: Pixel degradation, backlight inconsistencies
- **Signal processing**: Display scaling, video compression, interface noise

## Cascading Failure Modes

Unlike syntax errors in text that are localized, pixel corruption can cause:

- **Silent logic errors**: `x > 5` becomes `x > 6`, changing program behavior without failing
- **Type confusion**: `123` becomes `128`, causing wrong calculations
- **Control flow corruption**: `if` becomes `1f`, breaking program structure entirely

## Statistical Reliability Challenge

Even with a 99.9% per-character decode accuracy, a 100-character program has only (~90% × 99.9%^100) ≈ 37% chance of perfect decode. As programs grow larger, reliability drops exponentially.

## Recovery Complexity

When decode fails, you face difficult choices:

- **Halt execution**: System becomes unusable until manual intervention
- **Use previous version**: May execute stale code that doesn't match display
- **Attempt correction**: Risk introducing new errors or making wrong assumptions

## Comparison to Your Mitigations

Your implementation includes several mitigations:

- **Grid fonts**: Reduce character ambiguity 
- **CRC checks**: Detect corruption in control data
- **Temporal voting**: Average across multiple frames
- **Heatmap visualization**: Show decode confidence

These help but don't eliminate the fundamental issue: you're coupling program correctness to perfect visual fidelity, which is inherently unreliable.

## The Analog Amplification

When you transition to analog (camera + display), these challenges multiply:

- Camera focus variations
- Rolling shutter effects  
- Lighting changes
- Physical vibrations
- Lens distortions

This is why the hybrid approach I mentioned earlier could be valuable—using pixels for visualization and verification while maintaining a separate, reliable execution path from memory. You get the visual programming benefits without betting program correctness on pixel-perfect decode.

The refresh synchronization and temporal voting you're implementing are good engineering practices, but they're optimizations around the core fragility rather than solutions to it.
