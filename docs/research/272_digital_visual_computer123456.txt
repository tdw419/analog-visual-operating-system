## Visual Python Concept Evaluation

The Visual Python idea (code→pixels→decode→execute→pixels) is conceptually interesting but raises several technical concerns:

**Technical Challenges:**
- OCR reliability: Your grid font approach helps, but pixel-level code representation is inherently fragile
- Execution safety: Even with sandboxing, executing decoded pixel data introduces attack vectors
- Performance bottlenecks: The decode step adds latency that could impact real-time requirements
- Error propagation: A single pixel error could corrupt entire code blocks

**Alternative Consideration:**
Rather than decoding pixels back to executable code, consider pixels as a *display format* for code that's stored separately. This maintains visual representation while avoiding the fragility of pixel-to-execution coupling.

## PXOS Workbench Architecture Critique

Your HAL/Bridge pattern implementation is well-structured, but the technical report makes some assumptions that warrant scrutiny:

**AsyncIO Choice:**

- FastPlotLib vs Matplotlib: The performance claims need empirical validation with your specific data patterns
- PySerial-asyncio: Good choice, but ensure you've considered the additional complexity vs. threading approaches

**Testing Strategy Concerns:**
Visual regression testing with pixel-perfect comparisons can be brittle across different display configurations and operating systems. Consider hash-based or feature-based comparison methods as alternatives.

## Architectural Gaps

Several areas could use more development:

1. **Error Recovery:** How does the system handle cascade failures when the visual decode fails?

Your architectural thinking is thorough and shows good software engineering principles.     source_lines: list[str] = field(default_factory=lambda: ["1+2"])
    caret: tuple[int, int] = (0, 0)
    last_result: str = ""

def safe_eval(expr: str) -> str:
    """Safely evaluates a string expression with whitelisted functions."""
    allowed = {"math": math, "range": range, "sum": sum, "len": len, "min": min, "max": max}
    try:
        return str(eval(expr, {"__builtins__": {}}, allowed))
    except Exception as e:
        return f"Err: {str(e)[:COLS - 5]}"

def draw_pane(px: Px, pane_id: str, lines: list[str], color_fg: str, color_bg: str, caret=None):
    """Draws a pane's content to the Px object."""
    x, y = PANES[pane_id]
    px.rect(x - 2, y - 2, PANE_W + 4, PANE_H + 4, "#333333")
    px.rect(x, y, PANE_W, PANE_H, color_bg)
    for r, line in enumerate(lines[:ROWS]):
        draw_text(px._draw, x, y + r * CELL_H, line.ljust(COLS)[:COLS], color_fg, color_bg)
    if caret and pane_id == "A":
        cx, cy = x + caret[1] * CELL_W, y + caret[0] * CELL_H
        px.rect(cx, cy + CELL_H - 2, CELL_W, 2, "#00FF00")

def decode_pane(img: Image.Image, pane_id: str) -> list[str]:
    """Decodes a pane's content from a given image."""
    x, y = PANES[pane_id]
    return [sample_text(img, x, y + r * CELL_H).rstrip() for r in range(ROWS)]

def draw_heatmap(px: Px, truth: list[str], decoded: list[str]):
    """Draws a heatmap to visualize decode errors."""
    x, y = PANES["D"]
    px.rect(x, y, PANE_W, PANE_H, "#0E0F14")
    for r in range(ROWS):
        t = (truth[r] if r < len(truth) else "").ljust(COLS)
        d = (decoded[r] if r < len(decoded) else "").ljust(COLS)
        for c in range(COLS):
            cx, cy = x + c * CELL_W, y + r * CELL_H
            if t[c] == d[c]:
                px.rect(cx, cy, CELL_W, CELL_H, "#002200")
            else:
                px.rect(cx, cy, CELL_W, CELL_H, "#220000")

    px.clear("#0E0F14")

    # Pane Labels
    px.text(PANES["A"][0], PANES["A"][1] - 12, "A: Source (edit)", 12, "#A0FFA0")

def decode(px: Px, st: VPState) -> Dict:

    img = px._img
    ctrl = read_control(img)
    decoded_lines = decode_pane(img, "B")
    ctrl["code"] = "\n".join(decoded_lines)
    return ctrl

def update(st: VPState, ctrl: Dict) -> VPState:

    code = ctrl.get("code", "").strip()
    result = safe_eval(code) if code else ""
    return VPState(pc=st.pc + 1, source_lines=st.source_lines, caret=st.caret, last_result=result)

#!/usr/bin/env python3
# visual_python_mvp.py — Visual Python MVP (live edit → pixels → decode → exec → pixels)
import sys
import time
import math
from dataclasses import dataclass, field
from typing import Dict, List
from PIL import Image, ImageDraw, ImageStat
import tkinter as tk
from py2px import Px
from control_strip import draw_control, read_control
from visual_codec import draw_text, sample_text, CELL_W, CELL_H, COLS, ROWS

# Configuration
PIXEL_READ = True  # Decode from pixels only
W, H = 640, 360
GAP = 8
CTRL_H = 24
PANE_W = (W - GAP*3) // 2
PANE_H = (H - CTRL_H - GAP*3) // 2
FRAME_DIR = "out_vpy_mvp"

@dataclass
class VPState:
    pc: int = 0
    source_lines: List[str] = field(default_factory=lambda: ["1+2"])
    caret: tuple[int, int] = (0, 0)  # (row, col)
    last_result: str = ""

def safe_eval(expr: str) -> str:
    """Safely evaluate a Python expression with whitelisted builtins and timeout."""
    import signal
    def timeout_handler(signum, frame):
        raise TimeoutError("Evaluation timed out")
    signal.signal(signal.SIGALRM, timeout_handler)
    signal.alarm(1)  # 1-second timeout
    allowed = {
        "math": math, "range": range, "sum": sum, "len": len,
        "min": min, "max": max, "abs": abs, "pow": pow
    }
    try:
        result = eval(expr, {"__builtins__": {}}, allowed)
        signal.alarm(0)
        return str(result)[:COLS-5]
    except Exception as e:
        signal.alarm(0)
        return f"Err: {str(e)[:COLS-5]}"

def draw_pane(px: Px, x: int, y: int, lines: List[str], color_fg="#FFFFFF", color_bg="#1E2330", caret=None):
    """Draw a text pane with optional caret."""
    for r, line in enumerate(lines[:ROWS]):
        draw_text(px._draw, x, y + r * CELL_H, line.ljust(COLS)[:COLS], color_fg, color_bg)
    if caret:
        cx, cy = x + caret[1] * CELL_W, y + caret[0] * CELL_H
        px.rect(cx, cy + CELL_H - 2, CELL_W, 2, "#00FF00")  # Caret underline

def decode_pane(img: Image.Image, x: int, y: int) -> List[str]:
    """Decode a text pane from pixel data."""
    return [sample_text(img, x, y + r * CELL_H).rstrip() for r in range(ROWS)]

def draw_heatmap(px: Px, x: int, y: int, truth: List[str], decoded: List[str]):
    """Visualize decoding errors as a heatmap."""
    for r in range(ROWS):
        t = (truth[r] if r < len(truth) else "").ljust(COLS)
        d = (decoded[r] if r < len(decoded) else "").ljust(COLS)
        for c in range(COLS):
            cx, cy = x + c * CELL_W, y + r * CELL_H
            color = "#002200" if t[c] == d[c] else "#220000"  # Green for OK, red for mismatch
            px.rect(cx, cy, CELL_W, CELL_H, color)

def draw_frame(px: Px, st: VPState):
    """Draw the entire frame with all panes."""
    px.clear("#0E0F14")
    # Pane coordinates
    ax, ay = GAP, CTRL_H + GAP  # A: Source (editable truth)
    bx, by = ax + PANE_W + GAP, ay  # B: Pixelated Source
    cx, cy = ax, ay + PANE_H + GAP  # C: Result
    dx, dy = bx, cy  # D: Heatmap
    # Labels
    px.text(ax, ay-12, "A: Source (edit)", 12, "#A0FFA0")
    px.text(bx, by-12, "B: Pixelated (decoded)", 12, "#9BB4FF")
    px.text(cx, cy-12, "C: Result", 12, "#FFD966")
    px.text(dx, dy-12, "D: Decode Heatmap", 12, "#FF8080")
    # Draw panes
    draw_pane(px, ax, ay, st.source_lines, color_fg="#FFFFFF", color_bg="#1A1F2B", caret=st.caret)
    draw_pane(px, bx, by, st.source_lines, color_fg="#FFFFFF", color_bg="#1E2330")
    draw_pane(px, cx, cy, [st.last_result], color_fg="#FFD966" if "Err" not in st.last_result else "#FF8080", color_bg="#1A1F2B")
    # Control strip
    draw_control(px._draw, {"SCREEN_ID": 100, "PC": st.pc, "MAILBOX_IN": 0, "MAILBOX_OUT": 0})

def decode(px: Px, st: VPState) -> Dict:
    """Decode the frame (control strip + pixelated code)."""
    img = px._img
    ctrl = read_control(img)
    if not ctrl.get("_crc_ok", False):
        ctrl["err"] = "crc"
    ctrl["code"] = decode_pane(img, bx, by)[0] if PIXEL_READ else st.source_lines[0].strip()
    return ctrl

def update(st: VPState, ctrl: Dict) -> VPState:
    """Update state based on decoded frame."""
    code = ctrl.get("code", "").strip()
    result = safe_eval(code) if code else ""
    return VPState(pc=st.pc+1, source_lines=st.source_lines, caret=st.caret, last_result=result)

class Editor:
    """Handle live keyboard input."""
    def __init__(self, master, state: VPState):
        self.master = master
        self.state = state
        self.master.bind("<Key>", self.handle_key)

    def handle_key(self, event):
        ch = event.char
        r, c = self.state.caret
        line = self.state.source_lines[r]
        if ch.isprintable() and len(line) < COLS:
            self.state.source_lines[r] = line[:c] + ch + line[c:]
            self.state.caret = (r, min(c+1, COLS-1))
        elif event.keysym == "BackSpace" and c > 0:
            self.state.source_lines[r] = line[:c-1] + line[c:]
            self.state.caret = (r, c-1)
        elif event.keysym == "Left" and c > 0:
            self.state.caret = (r, c-1)
        elif event.keysym == "Right" and c < len(line):
            self.state.caret = (r, c+1)
        elif event.keysym == "Return":
            self.state.source_lines[r] = line[:COLS].strip()

def log_frame(st: VPState, ctrl: Dict, frame_idx: int):
    """Log frame data for replay and debugging."""
    import json
    import os
    log = {
        "frame": frame_idx,
        "pc": st.pc,
        "source": st.source_lines,
        "decoded": ctrl.get("code", ""),
        "result": st.last_result,
        "crc_ok": ctrl.get("_crc_ok", False)
    }
    os.makedirs(FRAME_DIR, exist_ok=True)
    with open(f"{FRAME_DIR}/log_{frame_idx:06d}.json", "w") as f:
        json.dump(log, f, indent=2)

def run_visual_python_mvp(frames=90, out_dir=FRAME_DIR):
    """Run the Visual Python MVP (batch or live)."""
    root = tk.Tk()
    root.withdraw()  # Hide main window
    px = Px(W, H, 60, out_dir)
    st = VPState(source_lines=["1+2"], caret=(0, 0))
    editor = Editor(root, st)
    
    # Pre-programmed code changes for batch mode
    code_changes = {40: "sum(range(5))", 80: "(3*7)+4"}
    
    if frames:
        for f in range(frames):
            if f in code_changes:
                st.source_lines[0] = code_changes[f]
                st.caret = (0, len(st.source_lines[0]))
            draw_frame(px, st)
            px.tick()
            ctrl = decode(px, st)
            assert ctrl.get("SCREEN_ID") == 100 and ctrl.get("_crc_ok", True), f"Frame {f} failed: {ctrl}"
            if PIXEL_READ and ctrl["code"].strip() != st.source_lines[0].strip():
                draw_heatmap(px, dx, dy, st.source_lines, [ctrl["code"]])
            st = update(st, ctrl)
            log_frame(st, ctrl, f)
    else:  # Live interactive loop
        def loop():
            draw_frame(px, st)
            px.tick()
            ctrl = decode(px, st)
            if PIXEL_READ and ctrl["code"].strip() != st.source_lines[0].strip():
                draw_heatmap(px, dx, dy, st.source_lines, [ctrl["code"]])
            st = update(st, ctrl)
            log_frame(st, ctrl, st.pc)
            root.after(16, loop)  # ~60Hz
        loop()
        root.mainloop()

if __name__ == "__main__":
    run_visual_python_mvp(frames=90)  # Batch mode
    # run_visual_python_mvp(frames=0)  # Uncomment for live interactive mode
