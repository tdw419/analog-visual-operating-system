# visual_calculator.py
# A safe, bytecode-driven visual calculator (Python 3.11+ friendly)
# - REPL with variables, Ans memory, :help / :vars / :history / :quit
# - Bytecode VM that supports BINARY_OP, COMPARE_OP, jumps, PRECALL/CALL, PUSH_NULL
# - Optional 4-pane PNG artifact renderer (uses Pillow if available)

from __future__ import annotations
import dis, io, sys, math, time, textwrap
from typing import Any, Dict, List

# -------- allowed functions / names (sandbox) --------
ALLOWED: Dict[str, Any] = {
    # math
    'sin': math.sin, 'cos': math.cos, 'tan': math.tan,
    'asin': math.asin, 'acos': math.acos, 'atan': math.atan,
    'sinh': math.sinh, 'cosh': math.cosh, 'tanh': math.tanh,
    'sqrt': math.sqrt, 'log': math.log, 'log10': math.log10, 'exp': math.exp,
    'pow': pow,
    'pi': math.pi, 'e': math.e, 'tau': math.tau,
    # safe builtins
    'abs': abs, 'round': round, 'min': min, 'max': max,
    'int': int, 'float': float, 'complex': complex,
    'len': len, 'sum': sum,
    # allow print (captured to stdout)
    'print': print,
}

class VisualCalcVM:
    """Tiny VM that interprets CPython bytecode for calculator-style inputs."""
    def __init__(self) -> None:
        self.stack: List[Any] = []
        self.vars: Dict[str, Any] = {}   # user variables
        self.ans: Any = None             # last result
        self.exec_log: List[Dict[str, Any]] = []  # for visualization

    # ---------- helpers ----------
    def _is_expr(self, s: str) -> bool:
        # crude heuristic: top-level '=' => treat as statements, otherwise expression
        head = s.split('#', 1)[0]
        return '=' not in head and not head.strip().startswith(('while ', 'for ', 'if ', 'def ', 'class ', 'import ', 'from '))

    def _jump_to(self, target_offset: int, offset_to_index: Dict[int, int], n: int) -> int:
        return offset_to_index.get(target_offset, n)

    # ---------- main runner ----------
    def run(self, src: str, mode: str = 'auto') -> tuple[Any, str]:
        if mode == 'auto':
            mode = 'eval' if self._is_expr(src) else 'exec'
        code = compile(src, '<calc>', mode)

        instructions = list(dis.get_instructions(code))
        offset_to_index = {ins.offset: i for i, ins in enumerate(instructions)}
        ip = 0
        n = len(instructions)

        # capture stdout
        old_stdout = sys.stdout
        sys.stdout = io.StringIO()
        self.exec_log.clear()
        self.stack.clear()

        try:
            while ip < n:
                ins = instructions[ip]
                op = ins.opname
                arg = ins.argval

                self.exec_log.append({'ip': ip, 'op': op, 'arg': arg, 'stack': list(self.stack)})

                # housekeeping
                if op in ('RESUME', 'CACHE', 'EXTENDED_ARG'):
                    ip += 1
                    continue

                # constants / names
                if op == 'LOAD_CONST':
                    self.stack.append(arg); ip += 1; continue

                if op in ('LOAD_NAME', 'LOAD_GLOBAL'):
                    name = arg
                    if name in self.vars:
                        self.stack.append(self.vars[name])
                    elif name in ALLOWED:
                        self.stack.append(ALLOWED[name])
                    elif name == '__builtins__':
                        raise NameError('access to __builtins__ is blocked')
                    elif name == 'Ans':
                        self.stack.append(self.ans)
                    else:
                        raise NameError(f"name '{name}' is not defined (or not allowed)")
                    ip += 1; continue

                if op == 'STORE_NAME':
                    self.vars[arg] = self.stack.pop()
                    ip += 1; continue

                # unary
                if op == 'UNARY_NEGATIVE':
                    self.stack.append(-self.stack.pop()); ip += 1; continue
                if op == 'UNARY_POSITIVE':
                    self.stack.append(+self.stack.pop()); ip += 1; continue
                if op == 'UNARY_NOT':
                    self.stack.append(not self.stack.pop()); ip += 1; continue

                # binary (3.11 BINARY_OP uses argrepr for symbol)
                if op == 'BINARY_OP':
                    b, a = self.stack.pop(), self.stack.pop()
                    sym = ins.argrepr  # expected '+', '-', '*', '/', '//', '%', '**', '@'
                    if   sym == '+': self.stack.append(a + b)
                    elif sym == '-': self.stack.append(a - b)
                    elif sym == '*': self.stack.append(a * b)
                    elif sym == '/': self.stack.append(a / b)
                    elif sym == '//': self.stack.append(a // b)
                    elif sym == '%': self.stack.append(a % b)
                    elif sym == '**': self.stack.append(a ** b)
                    elif sym == '@':
                        if hasattr(a, '__matmul__'):
                            self.stack.append(a @ b)
                        else:
                            raise TypeError('@ not supported')
                    else:
                        raise NotImplementedError(f'Unsupported BINARY_OP: {sym!r}')
                    ip += 1; continue

                # a few legacy BIN ops for compatibility (harmless if not used)
                if op == 'BINARY_ADD':
                    b, a = self.stack.pop(), self.stack.pop(); self.stack.append(a + b); ip += 1; continue
                if op == 'BINARY_MULTIPLY':
                    b, a = self.stack.pop(), self.stack.pop(); self.stack.append(a * b); ip += 1; continue
                if op == 'BINARY_TRUE_DIVIDE':
                    b, a = self.stack.pop(), self.stack.pop(); self.stack.append(a / b); ip += 1; continue

                # comparisons
                if op == 'COMPARE_OP':
                    b, a = self.stack.pop(), self.stack.pop()
                    cmp = ins.argrepr  # '==', '<', 'in', etc.
                    self.stack.append({
                        '==': a == b, '!=': a != b, '<': a < b, '<=': a <= b,
                        '>': a > b,  '>=': a >= b, 'in': a in b, 'not in': a not in b,
                        'is': a is b, 'is not': a is not b
                    }[cmp])
                    ip += 1; continue

                # jumps / control flow
                if op in ('POP_JUMP_FORWARD_IF_FALSE', 'POP_JUMP_BACKWARD_IF_FALSE'):
                    cond = self.stack.pop()
                    ip = self._jump_to(arg, offset_to_index, n) if not cond else ip + 1
                    continue

                if op in ('POP_JUMP_FORWARD_IF_TRUE', 'POP_JUMP_BACKWARD_IF_TRUE'):
                    cond = self.stack.pop()
                    ip = self._jump_to(arg, offset_to_index, n) if cond else ip + 1
                    continue

                if op in ('JUMP_FORWARD', 'JUMP_BACKWARD', 'JUMP_ABSOLUTE'):
                    ip = self._jump_to(arg, offset_to_index, n); continue

                # iteration (enables for-loops)
                if op == 'GET_ITER':
                    self.stack.append(iter(self.stack.pop())); ip += 1; continue

                if op == 'FOR_ITER':
                    it = self.stack[-1]
                    try:
                        nxt = next(it)
                        self.stack.append(nxt)
                        ip += 1
                    except StopIteration:
                        self.stack.pop()  # remove iterator
                        ip = self._jump_to(arg, offset_to_index, n)
                    continue

                # calls (3.11 calling convention)
                if op == 'PUSH_NULL':
                    self.stack.append(None); ip += 1; continue

                if op == 'PRECALL':
                    ip += 1; continue  # no-op in our VM

                if op == 'CALL':
                    argc = ins.arg
                    args = [self.stack.pop() for _ in range(argc)][::-1]
                    func = self.stack.pop()
                    if func is print:
                        print(*args)
                        self.stack.append(None)
                    elif func in ALLOWED.values() or func in self.vars.values():
                        self.stack.append(func(*args))
                    else:
                        raise PermissionError('function call not allowed')
                    ip += 1; continue

                # stack / return
                if op == 'POP_TOP':
                    if self.stack: self.stack.pop()
                    ip += 1; continue

                if op == 'RETURN_VALUE':
                    break

                # default
                raise NotImplementedError(f'Opcode not supported: {op}')

        finally:
            out = sys.stdout.getvalue()
            sys.stdout = old_stdout

        value = self.stack[-1] if self.stack else None
        # update Ans
        self.ans = value if value is not None else self.ans
        return value, out

Commands:
  :help       Show this help
  :vars       Show variables (and Ans)
  :history    Show recent inputs and results

Examples:
  2+3*4

def repl() -> None:
    vm = VisualCalcVM()
    history: List[tuple[str, Any]] = []
    print("Visual Python Calculator — type :help for help. Ctrl+C or :quit to exit.")
    while True:
        try:
            line = input("calc> ").strip()
        except (EOFError, KeyboardInterrupt):
            print("\nbye"); break
        if not line:
            continue
        if line.startswith(':'):
            cmd = line[1:].strip().lower()
            if cmd == 'help':
                print(HELP)
            elif cmd == 'vars':
                for k,v in vm.vars.items():
                    print(f"{k} = {v!r}")
                print(f"Ans = {vm.ans!r}")
            elif cmd == 'history':
                for i,(src,res) in enumerate(history[-12:], 1):
                    print(f"{i:02d}: {src} => {res!r}")
            elif cmd == 'quit':
                print("bye"); break
            else:
                print("unknown command; try :help")
            continue
        try:
            val, out = vm.run(line, mode='auto')
            if out.strip():
                print(out.strip())
            if val is not None:
                print(val)
            history.append((line, vm.ans if val is None else val))
        except Exception as e:
            print(f"Error: {e}")

if __name__ == '__main__':
    repl()

```python
class VisualCalcVM:
    """Tiny bytecode VM tuned for calculator use (Python 3.11+ friendly)."""
    def __init__(self):
        self.stack = []
        self.vars = {"Ans": None}
        self.exec_log = []
        self.last_result = None

    def _looks_like_expr(self, s):
        """Simple heuristic to distinguish expressions from statements."""
        return '=' not in s.strip().split('#', 1)[0]

    def _jump_to(self, target_offset, offset_to_index, nins):
        """Helper function to map bytecode offset to instruction index."""
        return offset_to_index.get(target_offset, nins)

    def run(self, code_str, mode="auto"):
        """Execute code_str. Returns (value, stdout_str)."""
        self.stack.clear()
        self.exec_log.clear()
        
        detected_mode = mode
        if mode == "auto":
            detected_mode = "eval" if self._looks_like_expr(code_str) else "exec"

        co = compile(code_str, "<calc>", detected_mode)

        offset_to_index = {ins.offset: i for i, ins in enumerate(instructions)}

        nins = len(instructions)

        old_stdout = sys.stdout
        sys.stdout = io.StringIO()

        try:
            while ip < nins:
                ins = instructions[ip]
                op = ins.opname
                self.exec_log.append((ip, op, ins.argval, list(self.stack)))

                if op in ("RESUME", "CACHE", "EXTENDED_ARG"):
                    ip += 1
                elif op == "LOAD_CONST":
                    self.stack.append(ins.argval); ip += 1
                elif op in ("LOAD_NAME", "LOAD_GLOBAL"):
                    name = ins.argval
                    if name in self.vars:
                        self.stack.append(self.vars[name])
                    elif name in ALLOWED:
                        self.stack.append(ALLOWED[name])
                    elif name == "__builtins__":
                        raise NameError("access to __builtins__ is not allowed")
                    else:
                        raise NameError(f"name '{name}' is not defined (or not allowed)")
                    ip += 1
                elif op == "STORE_NAME":
                    self.vars[ins.argval] = self.stack.pop(); ip += 1
                elif op == "UNARY_NEGATIVE":
                    self.stack.append(-self.stack.pop()); ip += 1
                elif op == "UNARY_POSITIVE":
                    self.stack.append(+self.stack.pop()); ip += 1
                elif op == "UNARY_NOT":
                    self.stack.append(not self.stack.pop()); ip += 1
                elif op == "BINARY_OP":
                    b, a = self.stack.pop(), self.stack.pop()
                    sym = ins.argrepr
                    if sym == '+': self.stack.append(a + b)
                    elif sym == '-': self.stack.append(a - b)
                    elif sym == '*': self.stack.append(a * b)
                    elif sym == '/': self.stack.append(a / b)
                    elif sym == '//': self.stack.append(a // b)
                    elif sym == '%': self.stack.append(a % b)
                    elif sym == '**': self.stack.append(a ** b)
                    elif sym == '@':
                        if hasattr(a, '__matmul__'): self.stack.append(a @ b)
                        else: raise TypeError("@ not supported for operands")
                    else: raise NotImplementedError(f"BINARY_OP {sym}")
                    ip += 1
                elif op == "BINARY_ADD":
                    b, a = self.stack.pop(), self.stack.pop(); self.stack.append(a + b); ip += 1
                elif op == "BINARY_MULTIPLY":
                    b, a = self.stack.pop(), self.stack.pop(); self.stack.append(a * b); ip += 1
                elif op == "COMPARE_OP":
                    b, a = self.stack.pop(), self.stack.pop()
                    cmp = ins.argrepr
                    res = {'==': a == b, '!=': a != b, '<': a < b, '<=': a <= b, '>': a > b, '>=': a >= b}[cmp]
                    self.stack.append(res); ip += 1
                elif op in ("POP_JUMP_FORWARD_IF_FALSE", "POP_JUMP_BACKWARD_IF_FALSE"):
                    cond = self.stack.pop()
                    ip = self._jump_to(ins.argval, offset_to_index, nins) if not cond else ip + 1
                elif op in ("POP_JUMP_FORWARD_IF_TRUE", "POP_JUMP_BACKWARD_IF_TRUE"):
                    cond = self.stack.pop()
                    ip = self._jump_to(ins.argval, offset_to_index, nins) if cond else ip + 1
                elif op in ("JUMP_FORWARD", "JUMP_BACKWARD", "JUMP_ABSOLUTE"):
                    ip = self._jump_to(ins.argval, offset_to_index, nins)
                elif op == "PUSH_NULL":
                    self.stack.append(None); ip += 1
                elif op == "PRECALL":
                    ip += 1
                elif op == "CALL":
                    argc = ins.arg
                    args = [self.stack.pop() for _ in range(argc)][::-1]
                    func = self.stack.pop()
                    if func is print:
                        print(*args); self.stack.append(None)
                    elif func in ALLOWED.values():
                        self.stack.append(func(*args))
                    else:
                        raise PermissionError("function call not allowed in calculator")
                    ip += 1
                elif op == "POP_TOP":
                    if self.stack: self.stack.pop(); ip += 1
                elif op == "RETURN_VALUE":
                    ip = nins
                else:
                    raise NotImplementedError(f"Opcode not supported: {op}")
        finally:
            out = sys.stdout.getvalue()
            sys.stdout = old_stdout
        
        value = self.stack[-1] if self.stack else None
        self.vars["Ans"] = value if value is not None else self.vars.get("Ans", None)
        self.last_result = value
        return value, out
```

import dis, operator, io, sys, textwrap, hashlib, json, time, math
from typing import List, Dict, Any

# -------- Whitelisted math functions --------
ALLOWED = {
    'sin': math.sin, 'cos': math.cos, 'tan': math.tan, 'sqrt': math.sqrt,
    'pi': math.pi, 'e': math.e, 'log': math.log, 'log10': math.log10,
    'exp': math.exp, 'factorial': math.factorial, 'abs': abs, 'round': round,
    'min': min, 'max': max, 'pow': pow
}

# -------- Calculator state --------
class CalculatorVM:
    def __init__(self):
        self.stack: List[Any] = []
        self.variables: Dict[str, Any] = {'M': 0}  # Initialize memory register
        self.execution_log: List[Dict] = []
        self.history: List[Dict[str, str]] = []
        self.ans: Any = None
        self.seq_num: int = 1
        self.last_output: str = ""

    def execute_bytecode(self, instructions: List[dis.Instruction]) -> str:
        """Execute bytecode in a sandboxed environment."""
        output = io.StringIO()
        old_stdout = sys.stdout
        sys.stdout = output
        offset_to_index = {ins.offset: i for i, ins in enumerate(instructions)}
        ip = 0
        try:
            while ip < len(instructions):
                instr = instructions[ip]
                self.execution_log.append({"instr": instr.opname, "arg": str(instr.argval), "stack": list(self.stack), "ip": ip})
                if instr.opname == "LOAD_CONST":
                    self.stack.append(instr.argval)
                    ip += 1
                elif instr.opname == "LOAD_NAME":
                    name = instr.argval
                    if name == "Ans":
                        self.stack.append(self.ans)
                    elif name in ALLOWED:
                        self.stack.append(ALLOWED[name])
                    else:
                        self.stack.append(self.variables.get(name, None))
                    ip += 1
                elif instr.opname == "STORE_NAME":
                    self.variables[instr.argval] = self.stack.pop()
                    ip += 1
                elif instr.opname == "BINARY_OP":
                    b, a = self.stack.pop(), self.stack.pop()
                    opstr = instr.argval
                    self.stack.append({
                        '+': a + b, '-': a - b, '*': a * b, '/': a / b,
                        '//': a // b, '%': a % b, '**': a ** b
                    }[opstr])
                    ip += 1
                elif instr.opname == "BINARY_ADD":
                    b, a = self.stack.pop(), self.stack.pop()
                    self.stack.append(a + b)
                    ip += 1
                elif instr.opname == "BINARY_MULTIPLY":
                    b, a = self.stack.pop(), self.stack.pop()
                    self.stack.append(a * b)
                    ip += 1
                elif instr.opname == "PRECALL":
                    ip += 1
                elif instr.opname == "CALL":
                    argc = instr.arg
                    args = [self.stack.pop() for _ in range(argc)][::-1]
                    func = self.stack.pop()
                    if func == print:
                        print(*args)
                        self.stack.append(None)
                    else:
                        self.stack.append(func(*args))
                    ip += 1
                elif instr.opname == "POP_TOP":
                    self.stack.pop()
                    ip += 1
                elif instr.opname == "RETURN_VALUE":
                    if self.stack:
                        self.ans = self.stack.pop()
                    break
                else:
                    raise NotImplementedError(f"Unsupported opcode: {instr.opname}")
        except Exception as e:
            self.last_output = f"Error: {e}"
            return self.last_output
        finally:
            self.last_output = output.getvalue().strip()
            sys.stdout = old_stdout
        return self.last_output

    def reset_log(self):
        self.execution_log = []

    def memory_op(self, op: str, value: float = None):
        """Handle memory operations (M+, M-, MR, MC)."""
        if op == "M+":
            self.variables['M'] += value if value is not None else self.ans
        elif op == "M-":
            self.variables['M'] -= value if value is not None else self.ans
        elif op == "MR":
            return self.variables['M']
        elif op == "MC":
            self.variables['M'] = 0

def to_visual_ir(py_src: str) -> List[dis.Instruction]:
    """Convert Python source to bytecode instructions."""
    try:
        code = compile(py_src, "<calc>", "exec")
        return list(dis.get_instructions(code))
    except Exception as e:
        return [{"opname": "ERROR", "argval": str(e)}]

def decode_tiles(instructions: List[dis.Instruction]) -> List[tuple]:
    """Decode tiles for drift detection."""
    return [(instr.opname, str(instr.argval) if instr.argval else "") for instr in instructions]

def compute_crc(data: str) -> str:
    """Compute CRC for source code."""
    return hashlib.md5(data.encode()).hexdigest()[:8]

```python
class VisualPythonVM:
    def __init__(self):
        self.stack = []
        self.namespace = {'Ans': 0}  # Initialize with 'Ans' variable
        self.execution_log = []
        self.current_ip = 0  # Instruction pointer
        self.last_result = None  # For Ans
        
        # Safe environment - whitelisted functions
        self.allowed_globals = {
            # Math functions
            'sin': math.sin, 'cos': math.cos, 'tan': math.tan,
            'asin': math.asin, 'acos': math.acos, 'atan': math.atan,
            'sinh': math.sinh, 'cosh': math.cosh, 'tanh': math.tanh,
            'sqrt': math.sqrt, 'log': math.log, 'log10': math.log10,
            'exp': math.exp, 'pow': math.pow, 'pi': math.pi, 'e': math.e,
            # Built-in functions
            'abs': abs, 'round': round, 'min': min, 'max': max,
            'int': int, 'float': float, 'str': str,
            'len': len, 'sum': sum, 'range': range, 'print': print
        }
        
    def execute_bytecode(self, instructions):
        """Execute Python bytecode instructions with support for jumps and 3.11+ opcodes"""
        output = io.StringIO()
        old_stdout = sys.stdout
        sys.stdout = output
        
        # Build offset to index mapping for jumps
        offset_to_index = {ins.offset: i for i, ins in enumerate(instructions)}
        
        self.current_ip = 0
        while self.current_ip < len(instructions):
            instr = instructions[self.current_ip]
            
            # Log the instruction for visualization
            self.execution_log.append({
                'instruction': instr.opname,
                'arg': instr.argval if hasattr(instr, 'argval') else instr.arg,
                'stack_before': list(self.stack),
                'ip': self.current_ip
            })
            
            # Execute the instruction
            self.execute_instruction(instr, instructions, offset_to_index)
            
        sys.stdout = old_stdout
        return output.getvalue()
    
    def jump_to_offset(self, target_offset, instructions, offset_to_index):
        """Helper function to jump to a specific offset"""
        return offset_to_index.get(target_offset, len(instructions))
    
    def execute_instruction(self, instruction, instructions, offset_to_index):
        """Execute a single instruction with support for 3.11+ opcodes and jumps"""
        opname = instruction.opname
        arg = instruction.argval if hasattr(instruction, 'argval') else instruction.arg
        
        # Handle comparison operations
        if opname == 'COMPARE_OP':
            b, a = self.stack.pop(), self.stack.pop()
            cmp = arg  # In 3.11+, argval is a string like '==', '<', etc.
            self.stack.append({
                '==': a == b, '!=': a != b,
                '<': a < b, '<=': a <= b,
                '>': a > b, '>=': a >= b,
                'in': a in b, 'not in': a not in b,
                'is': a is b, 'is not': a is not b
            }[cmp])
            self.current_ip += 1
            
        # Handle conditional jumps
        elif opname in ("POP_JUMP_FORWARD_IF_FALSE", "POP_JUMP_BACKWARD_IF_FALSE"):
            cond = self.stack.pop()
            if not cond:
                self.current_ip = self.jump_to_offset(arg, instructions, offset_to_index)
            else:
                self.current_ip += 1
                
        elif opname in ("POP_JUMP_FORWARD_IF_TRUE", "POP_JUMP_BACKWARD_IF_TRUE"):
            cond = self.stack.pop()
            if cond:
                self.current_ip = self.jump_to_offset(arg, instructions, offset_to_index)
            else:
                self.current_ip += 1
                
        # Handle unconditional jumps
        elif opname in ("JUMP_FORWARD", "JUMP_BACKWARD", "JUMP_ABSOLUTE"):
            self.current_ip = self.jump_to_offset(arg, instructions, offset_to_index)
            
        # Handle 3.11+ compact arithmetic
        elif opname == "BINARY_OP":
            b, a = self.stack.pop(), self.stack.pop()
            opstr = arg  # e.g. '+', '-', '*', '/', '//', '%', '**'
            self.stack.append({
                '+': a + b, '-': a - b, '*': a * b, '/': a / b,
                '//': a // b, '%': a % b, '**': a ** b,
                '@': a @ b if hasattr(a, '__matmul__') else (_ for _ in ()).throw(TypeError("@ not supported"))
            }[opstr])
            self.current_ip += 1
            
        # Handle 3.11+ function calls
        elif opname == "PUSH_NULL":
            self.stack.append(None)
            self.current_ip += 1
            
        elif opname == "PRECALL":
            # No-op for us, just advance
            self.current_ip += 1
            
        elif opname == "CALL":
            argc = arg  # number of positional args
            args = [self.stack.pop() for _ in range(argc)][::-1]
            func = self.stack.pop()
            if func is print:
                print(*args)
                self.last_result = args[0] if len(args) == 1 else args
                self.stack.append(None)
            elif func in self.allowed_globals.values():
                result = func(*args)
                self.last_result = result
                self.stack.append(result)
            else:
                self.stack.append(None)
            self.current_ip += 1
            
        # Original handlers for compatibility
        elif opname == 'LOAD_CONST':
            self.stack.append(arg)
            self.current_ip += 1
            
        elif opname == 'LOAD_NAME':
            if arg == 'Ans' and self.last_result is not None:
                self.stack.append(self.last_result)
            elif arg in self.namespace:
                self.stack.append(self.namespace[arg])
            elif arg in self.allowed_globals:
                self.stack.append(self.allowed_globals[arg])
            else:
                self.stack.append(None)
            self.current_ip += 1
            
        elif opname == 'STORE_NAME':
            self.namespace[arg] = self.stack.pop()
            self.current_ip += 1
            
        elif opname == 'BINARY_ADD':
            b, a = self.stack.pop(), self.stack.pop()
            self.stack.append(a + b)
            self.current_ip += 1
            
        elif opname == 'BINARY_SUBTRACT':
            b, a = self.stack.pop(), self.stack.pop()
            self.stack.append(a - b)
            self.current_ip += 1
            
        elif opname == 'BINARY_MULTIPLY':
            b, a = self.stack.pop(), self.stack.pop()
            self.stack.append(a * b)
            self.current_ip += 1
            
        elif opname == 'BINARY_TRUE_DIVIDE':
            b, a = self.stack.pop(), self.stack.pop()
            self.stack.append(a / b)
            self.current_ip += 1
            
        elif opname == 'CALL_FUNCTION':
            func = self.stack.pop()
            args = [self.stack.pop() for _ in range(arg)]
            if func in self.allowed_globals.values():
                result = func(*reversed(args))
                self.last_result = result
                self.stack.append(result)
            else:
                self.stack.append(None)
            self.current_ip += 1
            
        elif opname == 'POP_TOP':
            if self.stack:
                self.stack.pop()
            self.current_ip += 1
            
        elif opname == 'RETURN_VALUE':
            if self.stack:
                self.last_result = self.stack.pop()
            self.current_ip += 1
            
        # Handle loops
        elif opname == 'GET_ITER':
            iterable = self.stack.pop()
            self.stack.append(iter(iterable))
            self.current_ip += 1
            
        elif opname == 'FOR_ITER':
            iterator = self.stack[-1]
            try:
                self.stack.append(next(iterator))
                self.current_ip += 1
            except StopIteration:
                self.stack.pop()  # Remove iterator
                self.current_ip = self.jump_to_offset(arg, instructions, offset_to_index)
        else:
            # Skip unsupported opcodes
            self.current_ip += 1

def code_to_visual_tiles(code_obj):
    """Convert Python bytecode to visual tiles"""
    instructions = list(dis.get_instructions(code_obj))
    tiles = []
    
    for instr in instructions:
        if instr.opname in VISUAL_OPCODE_MAP:
            tile_info = VISUAL_OPCODE_MAP[instr.opname].copy()
            tile_info['opname'] = instr.opname
            if hasattr(instr, 'argval') and instr.argval is not None:
                tile_info['arg'] = str(instr.argval)
            else:
                tile_info['arg'] = ''
            tile_info['offset'] = instr.offset
            tile_info['is_jump_target'] = instr.is_jump_target
            tiles.append(tile_info)
    
    return tiles

class VisualCalculator:
    def __init__(self):
        self.vm = VisualPythonVM()
        self.history = []
        
    def evaluate(self, expression):
        """Evaluate a mathematical expression and return the result"""
        # Handle variable assignments
        if '=' in expression:
            var_name, expr = expression.split('=', 1)
            var_name = var_name.strip()
            expr = expr.strip()
            
            # Evaluate the expression
            result = self._evaluate_expression(expr)
            
            # Store the variable
            self.vm.namespace[var_name] = result
            return f"{var_name} = {result}"
        
        # Handle regular expressions
        return self._evaluate_expression(expression)
    
    def _evaluate_expression(self, expression):
        """Evaluate a mathematical expression"""
        # Handle Ans reference
        expression = expression.replace('Ans', '___ans___')
        
        # Create a safe namespace
        namespace = {
            '___ans___': self.vm.last_result,
            **self.vm.namespace,
            **self.vm.allowed_globals
        }
        
        try:
            # Compile and execute the expression
            code_obj = compile(expression, "<calculator>", "eval")
            result = eval(code_obj, namespace)
            
            # Store the result as Ans
            self.vm.last_result = result
            
            return result
        except Exception as e:
            return f"Error: {str(e)}"
    
    def save_4pane(self, expression, path):
        """Generate a 4-pane visualization for an expression"""
        # Convert expression to a full Python statement
        if '=' not in expression:
            code_str = f"result = {expression}"
        else:
            code_str = expression
            
        # Reset VM for clean execution
        self.vm.current_ip = 0
        self.vm.execution_log = []
        
        # Generate visualization
        return create_visualization(code_str, self.vm, path)

def calculator_repl():
    """Interactive calculator REPL"""
    calculator = VisualCalculator()
    print("Visual Python Calculator")
    print("Type ':help' for commands, ':quit' to exit")
    print("Examples: 2+3*4, sqrt(9), x=10, 2*Ans")
    
    while True:
        try:
            # Get user input
            user_input = input("calc> ").strip()
            
            # Handle commands
            if user_input.lower() == ':quit':
                print("Goodbye!")
                break
            elif user_input.lower() == ':help':
                print("Available commands:")
                print("  :help   - Show this help message")
                print("  :quit   - Exit the calculator")
                print("  :vars   - Show current variables")
                print("  :save   - Save a 4-pane visualization of the last expression")
                print("Examples:")
                print("  2+3*4          - Arithmetic expressions")
                print("  sqrt(9)        - Math functions")
                print("  x=10           - Variable assignment")
                print("  2*Ans          - Use last result")
                continue
            elif user_input.lower() == ':vars':
                print("Variables:")
                for var, value in calculator.vm.namespace.items():
                    print(f"  {var} = {value}")
                if calculator.vm.last_result is not None:
                    print(f"  Ans = {calculator.vm.last_result}")
                continue
            elif user_input.lower() == ':save':
                if calculator.history:
                    last_expr = calculator.history[-1]
                    filename = f"calc_{len(calculator.history)}.png"
                    calculator.save_4pane(last_expr, filename)
                    print(f"Saved visualization to {filename}")
                else:
                    print("No expression to visualize")
                continue
            
            # Skip empty input
            if not user_input:
                continue
            
            # Evaluate the expression
            result = calculator.evaluate(user_input)
            print(result)
            
            # Add to history
            calculator.history.append(user_input)
            
        except KeyboardInterrupt:
            print("\nGoodbye!")
            break
        except Exception as e:
            print(f"Error: {str(e)}")

if __name__ == "__main__":
    # Run the calculator REPL
    calculator_repl()
```

```python
from PIL import Image, ImageDraw, ImageFont, PngImagePlugin
import dis, io, sys, math, textwrap, hashlib, json, time
from typing import Dict, List, Any, Tuple

# --- Whitelisted functions ---
ALLOWED = {
    **{k: getattr(math, k) for k in dir(math) if not k.startswith('_')},
    'abs': abs, 'round': round, 'min': min, 'max': max, 'pow': pow,
    'int': int, 'float': float, 'len': len, 'sum': sum, 'print': print
}

# --- Bytecode color mapping ---
OPCODE_PALETTE = {
    "LOAD_CONST": "#3b82f6", "BINARY_OP": "#ef4444", "CALL": "#10b981",
    "LOAD_NAME": "#a855f7", "STORE_NAME": "#f97316", "POP_TOP": "#f59e0b",
    "COMPARE_OP": "#8b5cf6", "JUMP_*": "#f43f5e", "PRECALL": "#94a3b8"
}

class VisualCalcVM:
    def __init__(self):
        self.stack: List[Any] = []
        self.vars: Dict[str, Any] = {"Ans": None}
        self.exec_log: List[Tuple] = []
        self.history: List[Dict] = []
        self.seq_num = 1

    def execute(self, code_str: str) -> Tuple[Any, str]:
        """Execute code_str in sandboxed VM. Returns (result, stdout)."""
        try:
            co = compile(code_str, "<calc>", "eval" if self._is_expr(code_str) else "exec")
            instructions = list(dis.get_instructions(co))
            offset_map = {ins.offset: i for i, ins in enumerate(instructions)}
            ip = 0
            old_stdout = sys.stdout
            sys.stdout = io.StringIO()

            while ip < len(instructions):
                ins = instructions[ip]
                self.exec_log.append((ip, ins.opname, ins.argval))
                self._execute_instruction(ins, instructions, offset_map, ip)
                ip += 1

            output = sys.stdout.getvalue()
            sys.stdout = old_stdout
            result = self.stack[-1] if self.stack else None
            self.vars["Ans"] = result
            return result, output
        except Exception as e:
            sys.stdout = old_stdout
            return None, f"Error: {str(e)}"

    def _is_expr(self, s: str) -> bool:
        return '=' not in s.split('#')[0].strip()

    def _execute_instruction(self, ins, instructions, offset_map, ip):
        op = ins.opname
        arg = ins.argval if hasattr(ins, 'argval') else ins.arg

        # --- Constants & Names ---
        if op == "LOAD_CONST":
            self.stack.append(ins.argval)
        elif op in ("LOAD_NAME", "LOAD_GLOBAL"):
            name = ins.argval
            if name == "Ans":
                self.stack.append(self.vars["Ans"])
            elif name in ALLOWED:
                self.stack.append(ALLOWED[name])
            elif name in self.vars:
                self.stack.append(self.vars[name])
            else:
                raise NameError(f"'{name}' not defined")
        elif op == "STORE_NAME":
            self.vars[ins.argval] = self.stack.pop()

        # --- Arithmetic (3.11+) ---
        elif op == "BINARY_OP":
            b, a = self.stack.pop(), self.stack.pop()
            ops = {
                '+': a + b, '-': a - b, '*': a * b, '/': a / b,
                '//': a // b, '%': a % b, '**': a ** b
            }
            self.stack.append(ops[ins.argrepr])

        # --- Comparisons & Jumps ---
        elif op == "COMPARE_OP":
            b, a = self.stack.pop(), self.stack.pop()
            cmp = ins.argrepr
            self.stack.append({
                '==': a == b, '!=': a != b, '<': a < b, '<=': a <= b,
                '>': a > b, '>=': a >= b
            }[cmp])
        elif op in ("POP_JUMP_IF_FALSE", "POP_JUMP_IF_TRUE"):
            cond = self.stack.pop()
            if (op == "POP_JUMP_IF_FALSE" and not cond) or (op == "POP_JUMP_IF_TRUE" and cond):
                ip = offset_map[ins.argval]

        # --- Function Calls ---
        elif op == "CALL":
            argc = ins.arg
            args = [self.stack.pop() for _ in range(argc)][::-1]
            func = self.stack.pop()
            if func is print:
                print(*args)
                self.stack.append(None)
            elif func in ALLOWED.values() or func in self.vars.values():
                self.stack.append(func(*args))
            else:
                raise PermissionError("Function not allowed")

        # --- Stack Management ---
        elif op == "POP_TOP":
            self.stack.pop()

    def render_4pane(self, code_str: str, output_path: str):
        """Generate 4-pane visualization for a calculation."""
        instructions = list(dis.get_instructions(compile(code_str, "<calc>", "eval")))
        img = Image.new("RGB", (1400, 900), "#0a0f1a")
        d = ImageDraw.Draw(img)
        try:
            font_code = ImageFont.truetype("DejaVuSansMono.ttf", 24)
            font_ui = ImageFont.truetype("DejaVuSans.ttf", 16)
        except:
            font_code = ImageFont.load_default()
            font_ui = ImageFont.load_default()

        # --- Draw Panes ---
        def pane(x, y, w, h, title):
            d.rounded_rectangle([x, y, x+w, y+h], radius=14, fill="#0b1220", outline="#1f2a44", width=2)
            d.text((x+14, y-24), title, fill="#a5b4fc", font=font_ui)
            return (x, y, x+w, y+h)

        gap, w, h = 18, 650, 400
        P = [pane(gap, gap*2, w, h, f"Pane {i+1} — {title}") for i, title in enumerate([
            "Source", "Pixelized Text", "Bytecode Tiles", "Result"
        ])]

        # --- Pane 1: Source ---
        y = P[0][1] + 20
        for line in textwrap.wrap(code_str, width=40):
            d.text((P[0][0]+20, y), line, fill="#e2e8f0", font=font_code)
            y += 34

        # --- Pane 2: Pixelized Text ---
        cell_w, cell_h = 18, 24
        gx, gy = P[1][0] + 16, P[1][1] + 16
        for r, line in enumerate(code_str.splitlines() or ['']):
            for c, ch in enumerate(line):
                cx, cy = gx + c*cell_w, gy + r*cell_h
                d.rectangle([cx, cy, cx+cell_w-2, cy+cell_h-2], fill="#1e293b")
                d.text((cx+3, cy+2), ch, fill="#c7d2fe", font=font_ui)

        # --- Pane 3: Bytecode Tiles ---
        tile_size = 48
        cols = max(1, (w-40)//(tile_size+6))
        ox, oy = P[2][0]+20, P[2][1]+24
        for i, ins in enumerate(instructions):
            cx = ox + (i%cols)*(tile_size+6)
            cy = oy + (i//cols)*(tile_size+6)
            color = OPCODE_PALETTE.get(ins.opname, "#475569")
            d.rounded_rectangle([cx, cy, cx+tile_size, cy+tile_size], radius=10, fill=color)
            label = ins.opname if ins.argval is None else f"{ins.opname}\n{ins.argval}"
            d.text((cx+tile_size/2, cy+tile_size/2), label, anchor="mm", fill="#ffffff", font=font_ui)

        # --- Pane 4: Result ---
        y = P[3][1] + 20
        result, _ = self.execute(code_str)
        d.text((P[3][0]+20, y), f"Result: {result}", fill="#facc15", font=font_code)
        y += 30
        d.text((P[3][0]+20, y), f"Ans: {self.vars['Ans']}", fill="#22c55e", font=font_code)

        img.save(output_path)
        return output_path

    def repl(self):
        """Interactive calculator REPL."""
        print("Visual Python Calculator (type :help for commands)")
        while True:
            try:
                line = input("> ").strip()
                if not line:
                    continue
                if line == ":quit":
                    break
                elif line == ":help":
                    print("Commands: :vars, :history, :quit")
                    print("Examples: 2+3*4, sin(pi/2), x=10, 2*Ans")
                    continue
                elif line == ":vars":
                    print("Variables:", self.vars)
                    continue
                elif line == ":history":
                    for i, entry in enumerate(self.history[-5:], 1):
                        print(f"{i}. {entry['input']} => {entry['output']}")
                    continue

                result, output = self.execute(line)
                self.history.append({"input": line, "output": output or str(result)})
                if output:
                    print(output)
                if result is not None:
                    print(result)
                self.render_4pane(line, f"calc_{self.seq_num}.png")
                self.seq_num += 1
            except KeyboardInterrupt:
                print("\nExiting...")
                break
            except Exception as e:
                print(f"Error: {e}")
```

```python
from PIL import Image, ImageDraw, ImageFont
import dis
import io
import sys
import math
import textwrap

# Define our visual instruction set - mapping bytecode to visual elements
VISUAL_OPCODE_MAP = {
    'LOAD_CONST': {'color': '#3B82F6', 'symbol': '⚡'},
    'LOAD_NAME': {'color': '#10B981', 'symbol': '📛'},
    'STORE_NAME': {'color': '#F59E0B', 'symbol': '💾'},
    'BINARY_ADD': {'color': '#EF4444', 'symbol': '➕'},
    'BINARY_SUBTRACT': {'color': '#8B5CF6', 'symbol': '➖'},
    'BINARY_MULTIPLY': {'color': '#EC4899', 'symbol': '✖️'},
    'BINARY_TRUE_DIVIDE': {'color': '#06B6D4', 'symbol': '➗'},
    'BINARY_OP': {'color': '#8B5CF6', 'symbol': '⚙️'},  # 3.11+ compact arithmetic
    'CALL_FUNCTION': {'color': '#F97316', 'symbol': '📞'},
    'PRECALL': {'color': '#A855F7', 'symbol': '⏳'},  # 3.11+ pre-call
    'POP_TOP': {'color': '#64748B', 'symbol': '🗑️'},
    'RETURN_VALUE': {'color': '#A855F7', 'symbol': '🔙'},
    'COMPARE_OP': {'color': '#8B5CF6', 'symbol': '⚖️'},
    'POP_JUMP_FORWARD_IF_FALSE': {'color': '#F43F5E', 'symbol': '↩️'},
    'POP_JUMP_BACKWARD_IF_FALSE': {'color': '#F43F5E', 'symbol': '↩️'},
    'POP_JUMP_FORWARD_IF_TRUE': {'color': '#F43F5E', 'symbol': '↪️'},
    'POP_JUMP_BACKWARD_IF_TRUE': {'color': '#F43F5E', 'symbol': '↪️'},
    'JUMP_FORWARD': {'color': '#F43F5E', 'symbol': '⏭️'},
    'JUMP_BACKWARD': {'color': '#F43F5E', 'symbol': '⏮️'},
    'JUMP_ABSOLUTE': {'color': '#F43F5E', 'symbol': '⏩'},
    'PUSH_NULL': {'color': '#94A3B8', 'symbol': '⬜'}  # 3.11+ function call prep
}

class VisualPythonVM:
    def __init__(self):
        self.stack = []
        self.namespace = {}
        self.execution_log = []
        self.current_ip = 0
        self.last_result = None
        
    def execute_bytecode(self, instructions):
        """Execute Python bytecode instructions with support for jumps and 3.11+ opcodes"""
        output = io.StringIO()
        old_stdout = sys.stdout
        sys.stdout = output
        
        # Build offset to index mapping for jumps
        offset_to_index = {ins.offset: i for i, ins in enumerate(instructions)}
        
        while self.current_ip < len(instructions):
            instr = instructions[self.current_ip]
            
            # Log the instruction for visualization
            self.execution_log.append({
                'instruction': instr.opname,
                'arg': instr.argval if hasattr(instr, 'argval') else instr.arg,
                'stack_before': list(self.stack),
                'ip': self.current_ip
            })
            
            # Execute the instruction
            self.execute_instruction(instr, instructions, offset_to_index)
            
        sys.stdout = old_stdout
        return output.getvalue()
    
    def jump_to_offset(self, target_offset, instructions, offset_to_index):
        """Helper function to jump to a specific offset"""
        return offset_to_index.get(target_offset, len(instructions))
    
    def execute_instruction(self, instruction, instructions, offset_to_index):
        """Execute a single instruction with support for 3.11+ opcodes and jumps"""
        opname = instruction.opname
        arg = instruction.argval if hasattr(instruction, 'argval') else instruction.arg
        
        # Handle comparison operations
        if opname == 'COMPARE_OP':
            b, a = self.stack.pop(), self.stack.pop()
            cmp = arg  # In 3.11+, argval is a string like '==', '<', etc.
            self.stack.append({
                '==': a == b, '!=': a != b,
                '<': a < b,  '<=': a <= b,
                '>': a > b,  '>=': a >= b,
                'in': a in b, 'not in': a not in b,
                'is': a is b, 'is not': a is not b
            }[cmp])
            self.current_ip += 1
            
        # Handle conditional jumps
        elif opname in ("POP_JUMP_FORWARD_IF_FALSE", "POP_JUMP_BACKWARD_IF_FALSE"):
            cond = self.stack.pop()
            if not cond:
                self.current_ip = self.jump_to_offset(arg, instructions, offset_to_index)
            else:
                self.current_ip += 1
                
        elif opname in ("POP_JUMP_FORWARD_IF_TRUE", "POP_JUMP_BACKWARD_IF_TRUE"):
            cond = self.stack.pop()
            if cond:
                self.current_ip = self.jump_to_offset(arg, instructions, offset_to_index)
            else:
                self.current_ip += 1
                
        # Handle unconditional jumps
        elif opname in ("JUMP_FORWARD", "JUMP_BACKWARD", "JUMP_ABSOLUTE"):
            self.current_ip = self.jump_to_offset(arg, instructions, offset_to_index)
            
        # Handle 3.11+ compact arithmetic
        elif opname == "BINARY_OP":
            b, a = self.stack.pop(), self.stack.pop()
            opstr = arg  # e.g. '+', '-', '*', '/', '//', '%', '**'
            self.stack.append({
                '+': a + b, '-': a - b, '*': a * b, '/': a / b,
                '//': a // b, '%': a % b, '**': a ** b,
                '@': a @ b if hasattr(a, '__matmul__') else (_ for _ in ()).throw(TypeError("@ not supported"))
            }[opstr])
            self.current_ip += 1
            
        # Handle 3.11+ function calls
        elif opname == "PUSH_NULL":
            self.stack.append(None)
            self.current_ip += 1
            
        elif opname == "PRECALL":
            # No-op for us, just advance
            self.current_ip += 1
            
        elif opname == "CALL":
            argc = arg  # number of positional args
            args = [self.stack.pop() for _ in range(argc)][::-1]
            func = self.stack.pop()
            if func is print:
                print(*args)
                self.last_result = args[0] if len(args) == 1 else args
                self.stack.append(None)
            else:
                result = func(*args)
                self.last_result = result
                self.stack.append(result)
            self.current_ip += 1
            
        # Original handlers for compatibility
        elif opname == 'LOAD_CONST':
            self.stack.append(arg)
            self.current_ip += 1
            
        elif opname == 'LOAD_NAME':
            if arg == 'Ans' and self.last_result is not None:
                self.stack.append(self.last_result)
            else:
                self.stack.append(self.namespace.get(arg, None))
            self.current_ip += 1
            
        elif opname == 'STORE_NAME':
            self.namespace[arg] = self.stack.pop()
            self.current_ip += 1
            
        elif opname == 'BINARY_ADD':
            b, a = self.stack.pop(), self.stack.pop()
            self.stack.append(a + b)
            self.current_ip += 1
            
        elif opname == 'BINARY_SUBTRACT':
            b, a = self.stack.pop(), self.stack.pop()
            self.stack.append(a - b)
            self.current_ip += 1
            
        elif opname == 'BINARY_MULTIPLY':
            b, a = self.stack.pop(), self.stack.pop()
            self.stack.append(a * b)
            self.current_ip += 1
            
        elif opname == 'BINARY_TRUE_DIVIDE':
            b, a = self.stack.pop(), self.stack.pop()
            self.stack.append(a / b)
            self.current_ip += 1
            
        elif opname == 'CALL_FUNCTION':
            func = self.stack.pop()
            args = [self.stack.pop() for _ in range(arg)]
            result = func(*reversed(args))
            self.last_result = result
            self.stack.append(result)
            self.current_ip += 1
            
        elif opname == 'POP_TOP':
            self.stack.pop()
            self.current_ip += 1

def code_to_visual_tiles(code_obj):
    """Convert Python bytecode to visual tiles"""

                tile_info['arg'] = str(instr.argval)
            else:
                tile_info['arg'] = ''
            tile_info['offset'] = instr.offset
            tile_info['is_jump_target'] = instr.is_jump_target
            tiles.append(tile_info)
    
    return tiles

class VisualCalculator:
    def __init__(self):
        self.vm = VisualPythonVM()
        self.history = []
        self.allowed_names = {
            # Math functions
            'sin': math.sin, 'cos': math.cos, 'tan': math.tan,
            'asin': math.asin, 'acos': math.acos, 'atan': math.atan,
            'sinh': math.sinh, 'cosh': math.cosh, 'tanh': math.tanh,
            'sqrt': math.sqrt, 'log': math.log, 'log10': math.log10,
            'exp': math.exp, 'pow': math.pow,
            'pi': math.pi, 'e': math.e,
            # Built-in functions
            'abs': abs, 'round': round, 'min': min, 'max': max,
            'int': int, 'float': float, 'str': str,
            'len': len, 'sum': sum
        }
        
    def evaluate(self, expression):
        """Evaluate a mathematical expression and return the result"""
        # Handle variable assignments
        if '=' in expression:
            var_name, expr = expression.split('=', 1)
            var_name = var_name.strip()
            expr = expr.strip()
            
            # Evaluate the expression
            result = self._evaluate_expression(expr)
            
            # Store the variable
            self.vm.namespace[var_name] = result
            return f"{var_name} = {result}"
        
        # Handle regular expressions
        return self._evaluate_expression(expression)
    
    def _evaluate_expression(self, expression):
        """Evaluate a mathematical expression"""
        # Handle Ans reference
        expression = expression.replace('Ans', '___ans___')
        
        # Create a safe namespace
        namespace = {
            '___ans___': self.vm.last_result,
            **self.vm.namespace,
            **self.allowed_names
        }
        
        try:
            # Compile and execute the expression
            code_obj = compile(expression, "<calculator>", "eval")
            result = eval(code_obj, namespace)
            
            # Store the result as Ans
            self.vm.last_result = result
            
            return result
        except Exception as e:
            return f"Error: {str(e)}"
    
    def save_4pane(self, expression, path):
        """Generate a 4-pane visualization for an expression"""
        # Convert expression to a full Python statement
        if '=' not in expression:
            code_str = f"result = {expression}"
        else:
            code_str = expression
            
        # Reset VM for clean execution
        self.vm.current_ip = 0
        self.vm.execution_log = []
        
        # Generate visualization
        return create_visualization(code_str, self.vm, path)

def calculator_repl():
    """Interactive calculator REPL"""
    calculator = VisualCalculator()
    print("Visual Python Calculator")
    print("Type ':help' for commands, ':quit' to exit")
    print("Examples: 2+3*4, sqrt(9), x=10, 2*Ans")
    
    while True:
        try:
            # Get user input
            user_input = input("calc> ").strip()
            
            # Handle commands
            if user_input.lower() == ':quit':
                print("Goodbye!")
                break
            elif user_input.lower() == ':help':
                print("Available commands:")
                print("  :help   - Show this help message")
                print("  :quit   - Exit the calculator")
                print("  :vars   - Show current variables")
                print("  :save   - Save a 4-pane visualization of the last expression")
                print("Examples:")
                print("  2+3*4          - Arithmetic expressions")
                print("  sqrt(9)        - Math functions")
                print("  x=10           - Variable assignment")
                print("  2*Ans          - Use last result")
                continue
            elif user_input.lower() == ':vars':
                print("Variables:")
                for var, value in calculator.vm.namespace.items():
                    print(f"  {var} = {value}")
                if calculator.vm.last_result is not None:
                    print(f"  Ans = {calculator.vm.last_result}")
                continue
            elif user_input.lower() == ':save':
                if calculator.history:
                    last_expr = calculator.history[-1]
                    filename = f"calc_{len(calculator.history)}.png"
                    calculator.save_4pane(last_expr, filename)
                    print(f"Saved visualization to {filename}")
                else:
                    print("No expression to visualize")
                continue
            
            # Skip empty input
            if not user_input:
                continue
            
            # Evaluate the expression
            result = calculator.evaluate(user_input)
            print(result)
            
            # Add to history
            calculator.history.append(user_input)
            
        except KeyboardInterrupt:
            print("\nGoodbye!")
            break
        except Exception as e:
            print(f"Error: {str(e)}")

if __name__ == "__main__":
    # Run the calculator REPL
    calculator_repl()
```

- **PUSH_NULL**: Handles null pushing for function calls
