def draw_text(draw: ImageDraw.ImageDraw, x: int, y: int, text: str, color_fg="#FFFFFF", color_bg="#000000"):
    cx, cy = x, y
    for ch in text[:COLS]:
        draw.rectangle([cx, cy, cx+CELL_W-1, cy+CELL_H-1], fill=color_bg)
        pts = GLYPH.get(ch, GLYPH[" "])
        for (px, py) in pts:
            draw.point((cx+px, cy+py), fill=color_fg)
        cx += CELL_W

def sample_text(img: Image.Image, x: int, y: int) -> str:
    # Convert a row of cells back into characters by matching the atlas.
    row = ""
    for c in range(COLS):
        x0, y0 = x + c*CELL_W, y
        # Build a fingerprint: collect on-pixels roughly where atlas would set them.
        on = []
        for ch, pts in GLYPH.items():
            # quick test: count bright points at atlas positions
            score = 0
            for (px, py) in pts[:max(1, len(pts))]:
                roi = img.crop((x0+px, y0+py, x0+px+1, y0+py+1))
                if ImageStat.Stat(roi).mean[0] > 127:
                    score += 1
            on.append((score, ch))
        on.sort(reverse=True)  # best score first
        row += on[0][1]
    return row.rstrip()

“Code area” is drawn with the grid font above.

# visual_python.py
from __future__ import annotations
from dataclasses import dataclass
from typing import Dict
from PIL import Image
from py2px import Px
from control_strip import draw_control, read_control
from visual_codec import draw_text, sample_text, CELL_W, CELL_H, COLS

PIXEL_READ = False  # Phase A: False (use truth tap). Flip to True to read text from pixels.

# Layout constants
W, H = 480, 240
CODE_X, CODE_Y = 8, 40
OUT_X, OUT_Y   = 8, 40 + CELL_H + 12

@dataclass
class VPState:
    pc: int = 0
    code_line: str = "1+2"
    last_result: str = ""

def safe_eval(expr: str) -> str:
    # Sandboxed: only literal math via eval with empty builtins; extend slowly.
    allowed_names = {}
    try:
        val = eval(expr, {"__builtins__": {}}, allowed_names)
        return str(val)
    except Exception as e:
        return f"Error: {e}"

def draw_frame(px: Px, st: VPState):
    px.clear("#0E0F14")
    # labels
    px.text(8, 16, "Visual Python — code → pixels → exec → pixels", 12, "#A0FFA0")
    px.text(8, 28, "CODE:", 12, "#9BB4FF")
    # code row
    draw_text(px._draw, CODE_X, CODE_Y, st.code_line.ljust(COLS)[:COLS], color_fg="#FFFFFF", color_bg="#1E2330")
    # output
    px.text(8, OUT_Y-12, "RESULT:", 12, "#FFD966")
    draw_text(px._draw, OUT_X, OUT_Y, st.last_result.ljust(COLS)[:COLS], color_fg="#FFFFFF", color_bg="#1A1F2B")
    # control strip
    draw_control(px._draw, {
        "SCREEN_ID": 100,            # fixed id for this screen
        "PC": st.pc & 0xFFF,
        "MAILBOX_IN": 0,
        "MAILBOX_OUT": 0,
    })

def decode(px: Px, st: VPState) -> Dict:
    img = px._img
    ctrl = read_control(img)
    if not ctrl.get("_crc_ok", False):
        ctrl["err"] = "crc"
    # Read back code line
    decoded_code = sample_text(img, CODE_X, CODE_Y) if PIXEL_READ else st.code_line
    ctrl["code"] = decoded_code.strip()
    return ctrl

def update(st: VPState, ctrl: Dict) -> VPState:
    code = ctrl.get("code", "")
    result = safe_eval(code) if code else ""
    return VPState(pc=st.pc+1, code_line=code, last_result=result)

def run_visual_python(frames=90, out_dir="out_vpy"):
    px = Px(W, H, 60, out_dir)
    st = VPState()

    for f in range(frames):
        # (Optional) mutate code mid-run to prove live round-trip
        if f == 30: st.code_line = "sum(range(5))"
        if f == 60: st.code_line = "(3*7) + 4"

        draw_frame(px, st)
        px.tick()

        ctrl = decode(px, st)
        # minimal acceptance: screen id + crc
        assert ctrl.get("SCREEN_ID") == 100 and ctrl.get("_crc_ok", False), f"accept failed: {ctrl}"
        st = update(st, ctrl)

if __name__ == "__main__":
    run_visual_python()

```python
# control_strip.py
from __future__ import annotations
from dataclasses import dataclass
from typing import Dict, List, Tuple
from PIL import Image, ImageDraw, ImageStat

# Layout: a 24px tall band at the top; bits as 4x4 cells with 1-cell spacers.
CELL = 4
MARGIN = 4
CTRL_HEIGHT = MARGIN*2 + CELL
CTRL_Y0 = 0

# Field order is part of the protocol. Keep stable.
FIELDS: List[Tuple[str, int]] = [
    ("SCREEN_ID", 12),
    ("PC", 12),
    ("MAILBOX_IN", 8),
    ("MAILBOX_OUT", 8),
    ("CRC16", 16),
]

def crc16_ccitt(buf: bytes, poly: int = 0x1021, init: int = 0xFFFF) -> int:
    c = init
    for b in buf:
        c ^= (b << 8) & 0xFFFF
        for _ in range(8):
            c = ((c << 1) ^ poly) & 0xFFFF if (c & 0x8000) else (c << 1) & 0xFFFF
    return c & 0xFFFF

def _pack_fields(fields: Dict[str, int]) -> bytes:
    # Pack all but CRC16 in order, MSB-first, then return bytes for CRC calc.
    bits: List[int] = []
    for name, nbits in FIELDS:
        if name == "CRC16":
            continue
        val = int(fields.get(name, 0)) & ((1 << nbits) - 1)
        for i in range(nbits-1, -1, -1):
            bits.append((val >> i) & 1)
    # to bytes
    out = bytearray()
    for i in range(0, len(bits), 8):
        byte = 0
        for j in range(8):
            if i+j < len(bits):
                byte = (byte << 1) | bits[i+j]
            else:
                byte <<= 1
        out.append(byte)
    return bytes(out)

def draw_control(draw: ImageDraw.ImageDraw, fields: Dict[str, int], *, x0: int = MARGIN, y0: int = CTRL_Y0 + MARGIN) -> None:
    # Compute CRC over non-CRC fields and paint all fields as bit cells.
    payload = _pack_fields(fields)
    crc = crc16_ccitt(payload)
    fields = dict(fields)
    fields["CRC16"] = crc

    x = x0
    for name, nbits in FIELDS:
        val = int(fields.get(name, 0)) & ((1 << nbits) - 1)
        for i in range(nbits-1, -1, -1):
            bit = (val >> i) & 1
            draw.rectangle([x, y0, x+CELL-1, y0+CELL-1],
                           fill=(255,255,255) if bit else (0,0,0))
            x += CELL
        x += CELL  # spacer

def read_control(img: Image.Image, *, x0: int = MARGIN, y0: int = CTRL_Y0 + MARGIN) -> Dict[str, int]:
    from math import floor
    res: Dict[str, int] = {}
    x = x0
    for name, nbits in FIELDS:
        val = 0
        for _ in range(nbits):
            roi = img.crop((x, y0, x+CELL, y0+CELL))
            mean = ImageStat.Stat(roi).mean[0]
            bit = 1 if mean > 127 else 0
            val = (val << 1) | bit
            x += CELL
        res[name] = val
        x += CELL  # spacer

    # Verify CRC
    provided = res.get("CRC16", 0)
    payload = _pack_fields(res)  # uses all fields except CRC16 internally
    expect = crc16_ccitt(payload)
    res["_crc_ok"] = (expect == provided)
    return res
```

Here’s how I’d structure it so different parts of the map can hold different “files” (program stacks) of wildly different heights, without breaking the metaphor:

---

### 1. Treat the map as a sparse, chunked address space
- **Global coordinates**: the whole map is an integer grid.
- **Chunking**: divide into fixed‑size regions (e.g. 256×256 px) so we can index and page efficiently.
- **Chunk index**: each chunk has a tiny manifest listing the stacks anchored there.

---

### 2. A “program” is just a stack anchored to a region
- **Stack header**: `{stack_id, program_id, mount_at, bbox, z, frames[], state}`
- **Frames**: immutable blobs (PNGs or tile‑sheets) with ECC/fiducials.
- **State**: PC, mailboxes, any runtime vars.
- **Mount point**: top‑left coordinate on the map.

---

### 3. Paging logic in the runtime
- **Viewport → chunks**: figure out which chunks are visible.
- **Query stacks**: pull only those whose bounding boxes intersect the viewport.
- **Advance visible stacks**: run their `read → decide → draw` loop for one frame.
- **Composite**: layer them into the framebuffer by z‑order.

---

### 4. Different “files” in different places
- **Placing**: `place_program(program_id, at=(x,y), z=…)` writes the stack header into the relevant chunk indices.
- **Moving**: update `mount_at`/`bbox` and re‑index.
- **Deleting**: remove from indices; content‑addressed frames remain if referenced elsewhere.
- **Multiple instances**: same program can be mounted in many places with independent state.

---

### 5. Why digital‑first makes this painless
- You can simulate the infinite map, chunk paging, and stack advancement entirely in software.
- You can stress‑test with hundreds of stacks of different heights without worrying about camera latency or display refresh.
- You can verify that the control strip and decode logic work no matter where a stack is placed.
- When you swap in the analog backend, the paging and stacking logic doesn’t change — only the `FrameSink`/`FrameSource` implementations do.

# How we lay the world out
* **Global coords:** 2‑D integer plane (…,−2,−1,0,1,2,…).
* **Chunking:** split space into fixed chunks, e.g. 256×256 px. This keeps paging/rendering O(visible).
* **Program = Stack:** a file with:
  * `program_id` (content‑hash),
  * `bbox` (x0,y0,x1,y1) where it lives on the map,
  * `frames` (list of immutable frames or keyframes + deltas),
  * `state` (PC, mailboxes…),
  * `mount_at` (top‑left coordinate the stack is anchored to).
* **Multiple programs per area:** Any chunk can hold N stacks; we z‑order by `layer` or `z`.
* **Files on the map:** The map *is* a filesystem. Think paths like `/map/{chunk_x}/{chunk_y}/index.json` listing the stacks inside that chunk; the stack files live at `/programs/{program_id}.pxstack`.

# How the loop works (high level)
1. **Viewport → chunks:** compute which chunks overlap the viewport.
2. **Query stacks:** for those chunks, list stacks whose `bbox` intersects the viewport.
3. **Advance only visible stacks:** for each stack, run `read → decide → draw` for *its next frame*.
4. **Composite:** render stacks into the framebuffer by z‑order.
5. **Persist deltas:** if the stack advanced, write back its new `state` and append frame refs (content‑addressed).

# Addressing + placement
* **Drop a program** at (x,y) by writing a stack header:
  ```json
  {
    "program_id": "sha256:…",
    "mount_at": {"x": 12345, "y": -222},
    "bbox": {"x0":12345,"y0":-222,"x1":12345+W,"y1":-222+H},
    "z": 10,
    "frames": ["blake3:…", "blake3:…"],
    "state": {"pc":0,"mailbox_out":0}
  }
  ```
* **Moving/resize** = update `mount_at`/`bbox` (and reindex chunk membership for edges).

# Storage choices (simple & robust)
* **Chunk index** (append‑only JSON): `/map/{cx}/{cy}/index.json` → array of `{stack_id, bbox, z}`.
* **Stack file**: `/stacks/{stack_id}.json` → header + pointers to frame blobs.
* **Frames**: `/frames/{hash}.png` or `.pxsheet` (tile+ECC format).
* **Content‑address all blobs** for dedup across programs.

```python
# map_runtime.py
from dataclasses import dataclass
from typing import List, Dict, Tuple, Iterable, Optional
from PIL import Image
CHUNK = 256
def chunk_of(x:int, y:int) -> Tuple[int,int]:
    return x // CHUNK if x>=0 else -((-x-1)//CHUNK)-1, y // CHUNK if y>=0 else -((-y-1)//CHUNK)-1
@dataclass
class BBox: x0:int; y0:int; x1:int; y1:int
@dataclass
class StackHeader:
    stack_id: str
    program_id: str
    bbox: BBox
    z: int
    state: Dict
    frames: List[str] # content hashes for frames or keyframes
class ChunkIndex:
    def __init__(self, store): self.store = store # simple KV (path->bytes)
    def list_stacks(self, cx:int, cy:int) -> List[str]:
        data = self.store.get_json(f"/map/{cx}/{cy}/index.json") or []
        return [e["stack_id"] for e in data]
    def upsert(self, cx:int, cy:int, entry:Dict): ...
    def remove(self, cx:int, cy:int, stack_id:str): ...
class StackStore:
    def __init__(self, store): self.store = store
    def load(self, stack_id:str) -> StackHeader:
        d = self.store.get_json(f"/stacks/{stack_id}.json")
        bb = BBox(**d["bbox"])
        return StackHeader(stack_id=d["stack_id"], program_id=d["program_id"], bbox=bb, z=d["z"], state=d["state"], frames=d["frames"])
    def save(self, hdr:StackHeader): ...
class FrameStore:
    def __init__(self, store): self.store = store
    def get_frame(self, h:str) -> Image.Image: return self.store.get_image(f"/frames/{h}.png")
    def put_frame(self, img:Image.Image) -> str:
        # hash->path, write if missing, return hash
        ...
class MapRuntime:
    def __init__(self, store, renderer, decoder, reducer):
        self.chunks = ChunkIndex(store)
        self.stacks = StackStore(store)
        self.frames = FrameStore(store)
        self.renderer = renderer # program_id -> draw(state) -> PIL.Image
        self.decoder = decoder # program_id -> decode(image) -> dict
        self.reducer = reducer # program_id -> update(decoded,state,inputs) -> new_state
    def stacks_in_view(self, view:BBox) -> List[StackHeader]:
        cxs = range(chunk_of(view.x0,0)[0], chunk_of(view.x1-1,0)[0]+1)
        cys = range(chunk_of(0,view.y0)[1], chunk_of(0,view.y1-1)[1]+1)
        seen, out = set(), []
        for cx in cxs:
            for cy in cys:
                for sid in self.chunks.list_stacks(cx, cy):
                    if sid in seen: continue
                    seen.add(sid)
                    hdr = self.stacks.load(sid)
                    if not (hdr.bbox.x1 <= view.x0 or hdr.bbox.x0 >= view.x1 or hdr.bbox.y1 <= view.y0 or hdr.bbox.y0 >= view.y1):
                        out.append(hdr)
        return sorted(out, key=lambda h: h.z)
    def step_visible(self, view:BBox, inputs:Dict) -> Image.Image:
        # 1) render + advance each visible stack
        layers: List[Tuple[int, Image.Image, StackHeader]] = []
        for hdr in self.stacks_in_view(view):
            img = self.renderer.draw(hdr.program_id, hdr.state, hdr.bbox) # full stack frame (local coords)
            decoded = self.decoder.decode(hdr.program_id, img)
            new_state = self.reducer.update(hdr.program_id, decoded, hdr.state, inputs)
            if new_state != hdr.state:
                hdr.state = new_state
                # optional: persist frame (content addressed)
                # h = self.frames.put_frame(img); hdr.frames.append(h); self.stacks.save(hdr)
            layers.append((hdr.z, img, hdr))
        # 2) composite in z-order into final framebuffer
        fb = Image.new("RGB", (view.x1-view.x0, view.y1-view.y0), "#000000")
        for _, img, hdr in layers:
            ox, oy = hdr.bbox.x0 - view.x0, hdr.bbox.y0 - view.y0
            fb.paste(img, (ox, oy), mask=None)
        return fb
```

# How you “put different files in different places”
* **Mounting:** `place_program(program_id, at=(x,y), z=…)` creates a `stack_id`, sets `bbox` from the program’s declared size, and writes it into all affected **chunk indices** (edges/corners too).
* **Multiple versions:** You can mount *the same* `program_id` many times (different `stack_id`s, states, positions).
* **Moving:** remove from old chunk indices, update `bbox`, re‑index in new chunks.
* **Deleting:** remove the `stack_id` from indices (the content‑addressed frames remain if referenced elsewhere).

# How “big stack = complex program” works
* The **stack’s length** (or keyframe count) grows as it runs. We never rewrite frames; we append.
* **Code = data on the map:** if a program compiles down to frame‑sheets (`.pxsheet`), the stack is literally its sheets.
* A tiny control strip on each frame carries `{PROGRAM_ID, STACK_ID, PC, CRC}`, so any future capture can **self‑locate** and resume.

# Nice extras (when you’re ready)
* **Quadtree index** for faster spatial queries (optional; chunking is usually enough).
* **Transactional writes** for stack/state updates (append‑only journal → compact).
* **Content addressing everywhere** (BLAKE3/xxh3) so identical frames across programs don’t duplicate storage.
* **Cross‑stack messaging** via a small, map‑scoped mailbox: `/bus/{x_chunk}_{y_chunk}.q` for local broadcasts.
* **Permissions**: a stack can declare “collides_with” policy (e.g., top‑most wins, blend, or occlude).

```python
import os
import time
from PIL import Image
from contracts import *

class FrameSinkDigital:
    """Digital implementation - saves frames to disk"""
    def __init__(self, output_dir: str = "frames"):
        self.output_dir = output_dir
        os.makedirs(output_dir, exist_ok=True)
        self.frame_count = 0
    
    def present(self, image: Image.Image) -> None:
        filename = os.path.join(self.output_dir, f"frame_{self.frame_count:06d}.png")
        image.save(filename)
        self.frame_count += 1

class FrameSourceDigital:
    """Digital implementation - reads frames from disk"""
    def __init__(self, input_dir: str = "frames"):
        self.input_dir = input_dir
        self.frame_count = 0
    
    def capture(self) -> Image.Image:
        filename = os.path.join(self.input_dir, f"frame_{self.frame_count:06d}.png")
        self.frame_count += 1
        return Image.open(filename)

class ClockDigital:
    """Digital clock implementation"""
    def __init__(self, fps: int = 60):
        self.fps = fps
        self.frame_time = 1.0 / fps
        self.last_frame_time = time.time()
    
    def sleep(self, frames: int) -> None:
        duration = frames * self.frame_time
        time.sleep(duration)
    
    def tick(self) -> None:
        # Maintain frame pacing
        current_time = time.time()
        elapsed = current_time - self.last_frame_time
        sleep_time = max(0, self.frame_time - elapsed)
        time.sleep(sleep_time)
        self.last_frame_time = time.time()
```

```python
        # Save all artifacts

        with open(log_path / "report.json", "w") as f:
            json.dump(report.__dict__, f, indent=2)
        
        # Save tile comparison data
        self._save_tile_comparison(original_tiles, decoded_tiles, log_path)

        comparison_data = []

                "tile_id": i,
                "original_payload": orig.payload.hex(),
                "decoded_payload": dec.payload.hex(),
                "ecc_corrected": orig.payload != dec.payload and dec.ecc_ok,
                "ecc_failed": orig.payload != dec.payload and not dec.ecc_ok
            })
        
        with open(log_path / "tile_comparison.json", "w") as f:
```

### **1. Core Contracts and Data Structures**

These define the fundamental building blocks of the infinite map.

#### **`map_contracts.py`**

```python
from __future__ import annotations
from dataclasses import dataclass, field
from typing import List, Dict, Tuple, Optional, Any, Protocol
from PIL import Image

@dataclass
class Point:
    x: int
    y: int

@dataclass
class BBox:
    x0: int
    y0: int
    x1: int
    y1: int

@dataclass
class ProgramState:
    pc: int = 0
    mailbox_out: int = 0

@dataclass
class StackHeader:
    stack_id: str
    program_id: str
    mount_at: Point
    bbox: BBox
    z: int
    state: ProgramState = field(default_factory=ProgramState)
    frames: List[str] = field(default_factory=list)

class Program(Protocol):
    id: str
    def draw(self, state: ProgramState, px: Px, local_bbox: BBox) -> None:...
    def decode(self, img: Image.Image) -> Dict[str, Any]:...
    def update(self, decoded: Dict[str, Any], state: ProgramState, inputs: Dict[str, Any]) -> Tuple]:...
    def accept(self, decoded: Dict[str, Any]) -> bool:...

# Frame/pixel abstraction for rendering
class Px(Protocol):
    width: int
    height: int
    _draw: ImageDraw.Draw
    def rect(self, x:int, y:int, w:int, h:int, color:str) -> None:...
    def line(self, x1:int, y1:int, x2:int, y2:int, width:int, color:str) -> None:...
    def text(self, x:int, y:int, text:str, size:int, color:str) -> None:...
    def clear(self, color:str) -> None:...
    def tick(self) -> None:...
```

### **3. The Map Runtime**

This orchestrates the entire system, managing programs on the infinite canvas and advancing them frame by frame.

#### **`map_runtime.py`**

```python
from __future__ import annotations
import json
import os
import time
from typing import List, Dict, Tuple, Iterable, Optional, Protocol
from PIL import Image, ImageDraw, ImageStat
from map_contracts import BBox, StackHeader, Program, ProgramState, Px
from toy_programs import REGISTRY

CHUNK_SIZE = 256

def chunk_of(x: int, y: int) -> Tuple[int, int]:
    return x // CHUNK_SIZE, y // CHUNK_SIZE

class Store(Protocol):
    def get(self, key: str) -> Optional[bytes]:...
    def put(self, key: str, value: bytes) -> None:...

class InMemStore(Store):
    """In-memory key-value store for development"""
    def __init__(self):
        self._data: Dict[str, bytes] = {}
    def get(self, key: str) -> Optional[bytes]:
        return self._data.get(key)
    def put(self, key: str, value: bytes) -> None:
        self._data[key] = value

class MapRuntime:
    def __init__(self, store: Store):
        self.store = store
        self._stacks: Dict = {} # In-memory cache for headers

    def stacks_in_view(self, view: BBox) -> List:
        # Simplified for in-memory store; a real version would use a chunk index
        visible_stacks =
        for header in self._stacks.values():
            if self._intersects(header.bbox, view):
                visible_stacks.append(header)
        return sorted(visible_stacks, key=lambda h: h.z)

    def _intersects(self, bbox1: BBox, bbox2: BBox) -> bool:
        return not (bbox1.x1 <= bbox2.x0 or bbox1.x0 >= bbox2.x1 or bbox1.y1 <= bbox2.y0 or bbox1.y0 >= bbox2.y1)

    def place_program(self, program_id: str, mount_at: Point, bbox: BBox, z: int) -> StackHeader:
        stack_id = str(uuid.uuid4())
        header = StackHeader(stack_id=stack_id, program_id=program_id, mount_at=mount_at, bbox=bbox, z=z)
        self._stacks[stack_id] = header
        return header

    def step_visible(self, view: BBox, inputs: Dict[str, Any]) -> Image.Image:
        # 1) Create a blank framebuffer for compositing
        framebuffer = Image.new("RGB", (view.x1 - view.x0, view.y1 - view.y0), "#000000")
        
        # 2) Render and advance each visible stack
        for hdr in self.stacks_in_view(view):
            program = REGISTRY.get(hdr.program_id)
            if not program:
                continue
            # Create a local image for the stack's frame
            img = Image.new("RGB", (hdr.bbox.x1 - hdr.bbox.x0, hdr.bbox.y1 - hdr.bbox.y0), "#000000")
            px_proxy = self._create_px_proxy(img)
            program.draw(hdr.state, px_proxy, hdr.bbox)
            # Decode and update state
            decoded = program.decode(img)
            new_state, next_program_id = program.update(decoded, hdr.state, inputs)
            if new_state!= hdr.state:
                hdr.state = new_state
                self._stacks[hdr.stack_id] = hdr # Persist
            if next_program_id and next_program_id!= hdr.program_id:
                hdr.program_id = next_program_id
                hdr.state = ProgramState() # Reset state on switch
            # Composite the frame into the framebuffer
            ox, oy = hdr.bbox.x0 - view.x0, hdr.bbox.y0 - view.y0
            framebuffer.paste(img, (ox, oy), mask=None)
        
        # 3) return final framebuffer
        return framebuffer

    def _create_px_proxy(self, img: Image.Image) -> Px:
        """Creates a minimal proxy that implements the Px protocol for drawing"""
        draw = ImageDraw.Draw(img)
        class PxProxy:
            width, height = img.size
            _draw = draw
            def rect(self, x:int, y:int, w:int, h:int, color:str) -> None:
                draw.rectangle([x, y, x+w-1, y+h-1], fill=color)
            def line(self, x1:int, y1:int, x2:int, y2:int, width:int, color:str) -> None:
                draw.line([x1, y1, x2, y2], fill=color, width=width)
            def text(self, x:int, y:int, text:str, size:int, color:str) -> None:
                # Placeholder font logic
                font = ImageFont.load_default()
                draw.text((x,y), text, fill=color, font=font)
            def clear(self, color:str) -> None:
                img.paste(Image.new("RGB", img.size, color))
                self._draw = ImageDraw.Draw(img)
            def tick(self) -> None: pass
        return PxProxy()

def run_map_demo(frames=120, out_dir="out_map"):
    os.makedirs(out_dir, exist_ok=True)
    store = InMemStore()
    runtime = MapRuntime(store=store)

    # Mount a hello screen at (10, 10) on the map
    runtime.place_program(
        program_id="hello_screen",
        mount_at=Point(10, 10),
        bbox=BBox(10, 10, 330, 210),
        z=1
    )

    # Mount a clock screen at (300, 200) on the map
    runtime.place_program(
        program_id="clock_screen",
        mount_at=Point(300, 200),
        bbox=BBox(300, 200, 620, 400),
        z=2
    )
    
    # Simulate a viewport pan
    view = BBox(0, 0, 320, 200)
    for f in range(frames):
        # Render visible stacks and composite into the framebuffer
        framebuffer = runtime.step_visible(view, inputs={})
        framebuffer.save(os.path.join(out_dir, f"frame_{f:06d}.png"))
        # Pan the view to reveal the clock screen
        view.x0 = int(f * 2)
        view.y0 = int(f * 1.5)
        view.x1 = view.x0 + 320
        view.y1 = view.y0 + 200

if __name__ == "__main__":
    import uuid
    run_map_demo()
```

### **Why This is the Right Path**

  - **Sparse Map Model**: This approach naturally handles an "infinite map" by only operating on programs that are currently within the `view` (a much smaller subset).
  - **Frame as a Stack**: The `StackHeader` and `ProgramState` model your idea of a program as a stack of frames. The `update` method lets each program internally manage its own keyframes, program counter (`pc`), and state, making it a self-contained unit.
  - **Content Addressing**: The use of `program_id` and `stack_id` allows you to have multiple instances of the same program running on the map without duplicating logic. The content-addressed frames (via hashes) prevent duplicating data, just as you envisioned.
  - **Digital Fidelity**: The `step_visible` function first renders each program's frame, then captures and decodes it, then advances the state. This loop perfectly simulates the analog cycle of `present -> capture -> decode -> decide`.
  - **Separation of Concerns**: The `MapRuntime` is a pure orchestrator. It doesn't know how to draw a `RECT` or how to decode a `control strip`—it delegates those tasks to the `Program` and decoder helpers, just as your HAL design requires.

This blueprint gives you a clean, modular way to handle an infinite map, multiple programs, and per-program state, all in a deterministic digital environment that’s ready for the analog world.
